\section*{Περίληψη}
\addcontentsline{toc}{section}{Περίληψη}

Η αναζήτηση κοντινότερων γειτόνων (\selectlanguage{english}$k$-Nearest Neighbor Search\selectlanguage{greek}) αποτελεί ένα από τα θεμελιώδη προβλήματα στα σύγχρονα συστήματα επεξεργασίας πληροφορίας. Σε εφαρμογές όπως η σημασιολογική αναζήτηση, η ανάκτηση-επαυξημένη παραγωγή (\selectlanguage{english}RAG -- Retrieval-Augmented Generation\selectlanguage{greek}) και τα συστήματα σύστασης, η εξαντλητική αναζήτηση (\selectlanguage{english}brute-force\selectlanguage{greek}) δεν κλιμακώνεται για εκατομμύρια διανύσματα υψηλής διάστασης. Ο αλγόριθμος \selectlanguage{english}HNSW (Hierarchical Navigable Small World)\selectlanguage{greek} αποτελεί σήμερα μία από τις κυρίαρχες μεθόδους προσεγγιστικής αναζήτησης (\selectlanguage{english}ANN\selectlanguage{greek}) για παραγωγικές εφαρμογές. Ωστόσο, παρουσιάζει ένα δομικό περιορισμό: η παράμετρος \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek}, που ελέγχει το υπολογιστικό κόστος ανά ερώτημα, ορίζεται ενιαία για το σύνολο των ερωτημάτων, ανεξάρτητα από τη δυσκολία τους.

Η παρούσα εργασία εξετάζει το ερώτημα κατά πόσο ο αλγόριθμος αναζήτησης μπορεί να προσαρμόζει δυναμικά το υπολογιστικό \selectlanguage{english}budget\selectlanguage{greek} ανά ερώτημα. Για τον σκοπό αυτό, υλοποιείται και αξιολογείται μια βασική (\selectlanguage{english}Baseline\selectlanguage{greek}) υλοποίηση \selectlanguage{english}HNSW\selectlanguage{greek} σε \selectlanguage{english}C++\selectlanguage{greek} με διεπαφή \selectlanguage{english}Python\selectlanguage{greek} μέσω \selectlanguage{english}pybind11\selectlanguage{greek}, καθώς και τρεις προσαρμοστικές παραλλαγές από τη βιβλιογραφία: η \selectlanguage{english}DARTH\selectlanguage{greek} χρησιμοποιεί μοντέλο \selectlanguage{english}LightGBM\selectlanguage{greek} για πρόωρο τερματισμό της αναζήτησης· η \selectlanguage{english}PiP\selectlanguage{greek} εφαρμόζει πρόωρο τερματισμό βάσει κορεσμού του συνόλου \selectlanguage{english}top-$k$\selectlanguage{greek}· η \selectlanguage{english}Ada-ef\selectlanguage{greek} επιλέγει \selectlanguage{english}offline\selectlanguage{greek} το κατάλληλο \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek} ανά ερώτημα μέσω βαθμονόμησης βάσει εκτιμώμενης ομοιότητας ερωτήματος. Ως σημείο αναφοράς χρησιμοποιείται η βιβλιοθήκη \selectlanguage{english}FAISS\selectlanguage{greek} με \selectlanguage{english}HNSW\selectlanguage{greek}.

Η πειραματική αξιολόγηση πραγματοποιείται σε πέντε σύνολα δεδομένων διαφορετικής κλίμακας και χαρακτηριστικών (\selectlanguage{english}SIFT Small, Fashion-MNIST, MS MARCO Text, GloVe-100, SIFT-1M\selectlanguage{greek}), με κύριες μετρικές το \selectlanguage{english}Recall@10\selectlanguage{greek} και τις ερωτήσεις ανά δευτερόλεπτο (\selectlanguage{english}QPS\selectlanguage{greek}). Τα αποτελέσματα δείχνουν ότι η \selectlanguage{english}DARTH\selectlanguage{greek} επιτυγχάνει μέτρια βελτίωση σε σύνολα με ετερογενή κατανομή δυσκολίας ερωτημάτων, ενώ υπολείπεται σε ομοιόμορφα κατανεμημένα σύνολα όπως το \selectlanguage{english}GloVe-100\selectlanguage{greek}. Η \selectlanguage{english}PiP\selectlanguage{greek} εμφανίζει κορεσμό ανάκλησης σε μεγάλα σύνολα, ανίκανη να ξεπεράσει τη \selectlanguage{english}Baseline\selectlanguage{greek}. Η \selectlanguage{english}Ada-ef\selectlanguage{greek}, αν και επιτυγχάνει υψηλότερο \selectlanguage{english}recall\selectlanguage{greek} σε ορισμένα σενάρια, λειτουργεί με διαφορετικό μέτρο απόστασης, γεγονός που δυσχεραίνει την απευθείας σύγκριση. Συνολικά, η εργασία αναδεικνύει ότι η αποτελεσματικότητα των προσαρμοστικών στρατηγικών εξαρτάται κρίσιμα από τα χαρακτηριστικά της κατανομής δεδομένων, και ότι οι χαμηλού επιπέδου βελτιστοποιήσεις υλοποίησης επιδρούν εξίσου σημαντικά στην απόδοση με τον ίδιο τον αλγόριθμο.

\section{Εισαγωγή}

\subsection{Κίνητρο και Πλαίσιο}

Στην εποχή των μεγάλων γλωσσικών μοντέλων και των συστημάτων τεχνητής νοημοσύνης, ένα από τα πιο θεμελιώδη υπολογιστικά προβλήματα είναι η αναζήτηση ομοιότητας σε μεγάλες συλλογές δεδομένων. Συστήματα όπως οι μηχανές σημασιολογικής αναζήτησης, τα συστήματα ανάκτησης-επαυξημένης παραγωγής (\selectlanguage{english}RAG -- Retrieval-Augmented Generation\selectlanguage{greek}), και τα συστήματα σύστασεων (\selectlanguage{english}recommendation systems\selectlanguage{greek}) βασίζονται στην ικανότητα να βρίσκουν γρήγορα τα πιο «κοντινά» αντικείμενα σε μία βάση δεδομένων.

Το κλειδί αυτής της λειτουργίας είναι οι \textbf{διανυσματικές αναπαραστάσεις} (\selectlanguage{english}vector embeddings\selectlanguage{greek}): κάθε αντικείμενο --- κείμενο, εικόνα, ήχος --- μετατρέπεται σε ένα διάνυσμα πραγματικών αριθμών $\mathbf{x} \in \mathbb{R}^d$, όπου $d$ η διάσταση (συνήθως από δεκάδες έως μερικές χιλιάδες). Αυτή η αριθμητική αναπαράσταση κωδικοποιεί τα σημασιολογικά χαρακτηριστικά του αντικειμένου: δύο αντικείμενα με παρόμοιο νόημα ή περιεχόμενο αντιστοιχούν σε διανύσματα που βρίσκονται κοντά στον $d$-διάστατο χώρο, σύμφωνα με κάποιο μέτρο απόστασης (π.χ. ευκλείδεια απόσταση ή ομοιότητα συνημιτόνου).

Σε αυτό το πλαίσιο, η βασική επιχειρησιακή ερώτηση είναι: \textit{δοθέντος ενός ερωτήματος $\mathbf{q}$, ποια είναι τα $k$ πιο κοντινά διανύσματα σε μία συλλογή $N$ στοιχείων;}. Το πρόβλημα αυτό ονομάζεται \textbf{αναζήτηση κοντινότερων γειτόνων} (\selectlanguage{english}$k$-Nearest Neighbor Search, $k$-NNS\selectlanguage{greek}) και εμφανίζεται στον πυρήνα εφαρμογών παραγωγικής κλίμακας.

\subsection{Το Πρόβλημα της Κλιμάκωσης}

Η εξαντλητική αναζήτηση --- υπολογισμός της απόστασης του ερωτήματος από κάθε ένα από τα $N$ διανύσματα της βάσης --- εγγυάται τη σωστή απάντηση αλλά έχει κόστος $\mathcal{O}(N \cdot d)$ ανά ερώτημα. Για συλλογές εκατομμυρίων διανυσμάτων υψηλής διάστασης, αυτό μεταφράζεται σε εκατοντάδες εκατομμύρια αριθμητικές πράξεις ανά ερώτημα --- αδύνατο να εκτελεστεί σε πραγματικό χρόνο σε παραγωγικά συστήματα που καλούνται να εξυπηρετήσουν χιλιάδες ερωτήματα ανά δευτερόλεπτο.

Για τον λόγο αυτό, η ερευνητική κοινότητα έχει αναπτύξει μεθόδους \textbf{κατά προσέγγιση αναζήτησης κοντινότερων γειτόνων} (\selectlanguage{english}Approximate Nearest Neighbor Search -- ANN\selectlanguage{greek}): αλγόριθμοι που επιστρέφουν αποτελέσματα όχι απαραίτητα βέλτιστα, αλλά επαρκώς ακριβή --- και πολύ πιο γρήγορα. Η ποιότητα των αποτελεσμάτων μετριέται με τη μετρική \textbf{Recall@k}: το ποσοστό των πραγματικών $k$ κοντινότερων γειτόνων που επέστρεψε ο αλγόριθμος. Τιμή $\text{Recall@k} = 1.0$ σημαίνει τέλεια ακρίβεια, ενώ στην πράξη αποδεκτές θεωρούνται τιμές στο εύρος $0.9$--$0.99$ ανάλογα με την εφαρμογή.

\subsection{Ο Αλγόριθμος HNSW και το Πρόβλημα της Σταθερής Παραμετροποίησης}

Ο αλγόριθμος \selectlanguage{english}\textbf{HNSW} (Hierarchical Navigable Small World)\selectlanguage{greek} \cite{malkov2018hnsw} αποτελεί σήμερα την επικρατούσα μέθοδο ANN σε παραγωγικά συστήματα. Βασίζεται στην κατασκευή ενός πολυεπίπεδου γράφου: τα δεδομένα οργανώνονται σε ιεραρχία επιπέδων, με τα ανώτερα να λειτουργούν ως γρήγορος χάρτης πλοήγησης και το κατώτερο να εκτελεί λεπτομερή τοπική αναζήτηση. Ο συνδυασμός αυτός επιτρέπει υψηλό Recall@k με σημαντικά λιγότερους υπολογισμούς απόστασης σε σχέση με την εξαντλητική μέθοδο.

Ωστόσο, ο \selectlanguage{english}HNSW\selectlanguage{greek} παρουσιάζει ένα δομικό αδύνατο σημείο: η παράμετρος \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek} --- που ελέγχει πόσο εκτεταμένη θα είναι η αναζήτηση στο κατώτερο επίπεδο --- ορίζεται εκ των προτέρων, ίδια για κάθε ερώτημα. Στην πράξη, ερωτήματα των οποίων οι κοντινότεροι γείτονες βρίσκονται σε πυκνή, καλά οριοθετημένη γειτονιά («εύκολα» ερωτήματα) απαιτούν πολύ λιγότερους υπολογισμούς για υψηλό \selectlanguage{english}recall\selectlanguage{greek} σε σχέση με ερωτήματα που οι γείτονές τους είναι διάσπαρτοι στον χώρο («δύσκολα» ερωτήματα). Μια σταθερή τιμή \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek} αναγκαστικά σπαταλά υπολογισμούς στα εύκολα ερωτήματα ή θυσιάζει \selectlanguage{english}recall\selectlanguage{greek} στα δύσκολα.

\subsection{Στόχοι της Εργασίας}

Η παρούσα εργασία εξετάζει το ερώτημα: \textit{μπορεί ο αλγόριθμος αναζήτησης να προσαρμόζει δυναμικά τον αριθμό των υπολογισμών ανά ερώτημα, ανάλογα με τη δυσκολία του;}

Για να απαντηθεί αυτό, μελετώνται και υλοποιούνται τρεις διαφορετικές προσαρμοστικές προσεγγίσεις που έχουν προταθεί στη βιβλιογραφία:
\begin{itemize}
    \item \selectlanguage{english}\textbf{DARTH}\selectlanguage{greek} \cite{chatzakis2025darth}: χρησιμοποιεί μοντέλο μηχανικής μάθησης (\selectlanguage{english}LightGBM\selectlanguage{greek}) για να προβλέπει πότε η αναζήτηση μπορεί να τερματιστεί νωρίς χωρίς απώλεια ακρίβειας.
    \item \selectlanguage{english}\textbf{PiP}\selectlanguage{greek} \cite{teofili2025pip}: τερματίζει νωρίς την αναζήτηση παρακολουθώντας τη σταθεροποίηση του συνόλου \selectlanguage{english}top-$k$\selectlanguage{greek} για $\Delta$ συνεχόμενες επαναλήψεις (\selectlanguage{english}patience\selectlanguage{greek}-βάσει κορεσμός).
    \item \selectlanguage{english}\textbf{Ada-ef}\selectlanguage{greek} \cite{zhang2026adaef}: εκτιμά τη δυσκολία κάθε ερωτήματος μέσω βαθμονόμησης \selectlanguage{english}offline\selectlanguage{greek} και επιλέγει αυτόματα την κατάλληλη τιμή \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek} ανά ερώτημα.
\end{itemize}

Οι τρεις αυτές μέθοδοι αξιολογούνται συστηματικά και συγκρίνονται με δύο σημεία αναφοράς: τη δική μας υλοποίηση \selectlanguage{english}Baseline-HNSW\selectlanguage{greek} (χωρίς προσαρμοστικό μηχανισμό) και τη \selectlanguage{english}faiss-hnsw \cite{faiss2017}\selectlanguage{greek} (βιβλιοθήκη παραγωγικής χρήσης). Η αξιολόγηση πραγματοποιείται σε πέντε σύνολα δεδομένων διαφορετικής κλίμακας και χαρακτηριστικών, με κύριες μετρικές το \selectlanguage{english}Recall@10\selectlanguage{greek} και τις ερωτήσεις ανά δευτερόλεπτο (\selectlanguage{english}QPS -- Queries Per Second\selectlanguage{greek}).

\subsection{Συνεισφορές της Εργασίας}
Συνοπτικά, οι κύριες συνεισφορές της παρούσας εργασίας είναι οι εξής:
\begin{itemize}
    \item Υλοποίηση \selectlanguage{english}Baseline HNSW\selectlanguage{greek} σε \selectlanguage{english}C++\selectlanguage{greek} με \selectlanguage{english}pybind11\selectlanguage{greek}, με καταγραφή υπολογιστικού κόστους ανά φάση κατασκευής και αναζήτησης.
    \item Υλοποίηση των \selectlanguage{english}DARTH\selectlanguage{greek}, \selectlanguage{english}PiP\selectlanguage{greek} και \selectlanguage{english}Ada-ef\selectlanguage{greek} σε κοινό πλαίσιο αξιολόγησης, εξασφαλίζοντας άμεση συγκρισιμότητα των αποτελεσμάτων.
    \item Ενοποιημένη αξιολόγηση σε πέντε σύνολα δεδομένων διαφορετικής κλίμακας και χαρακτηριστικών κατανομής.
    \item Συγκριτική μελέτη του συμβιβασμού ακρίβειας--κόστους (\selectlanguage{english}accuracy-cost tradeoff\selectlanguage{greek}) ανά μέθοδο, με ανάλυση συμπεριφοράς σε εύκολα και δύσκολα ερωτήματα.
    \item Ανάλυση της επίδρασης χαρακτηριστικών κατανομής δεδομένων στην αποτελεσματικότητα των προσαρμοστικών στρατηγικών αναζήτησης.
\end{itemize}

\section{Θεωρητικό Υπόβαθρο}
Το κεφάλαιο αυτό θέτει το θεωρητικό πλαίσιο που υποστηρίζει την πειραματική σύγκριση των επόμενων κεφαλαίων. Παρουσιάζεται ο τυπικός ορισμός του προβλήματος αναζήτησης κοντινότερων γειτόνων, οι κύριες οικογένειες ANN μεθόδων, και η λεπτομερής λειτουργία του HNSW --- συμπεριλαμβανομένου ψευδοκώδικα και ενός κατατοπιστικού παραδείγματος.
\subsection{Αναζήτηση Κοντινότερων Γειτόνων}

Τυπικά, το πρόβλημα ορίζεται ως εξής: δοθείσης μιας συλλογής $N$ διανυσμάτων $\mathcal{X} = \{\mathbf{x}_1, \ldots, \mathbf{x}_N\} \subset \mathbb{R}^d$ και ενός ερωτήματος $\mathbf{q} \in \mathbb{R}^d$, ζητούνται τα $k$ στοιχεία της $\mathcal{X}$ με τη μικρότερη απόσταση από το $\mathbf{q}$ βάσει κάποιας μετρικής $d(\cdot, \cdot)$. Τα δύο μέτρα που χρησιμοποιούνται πιο συχνά στην πράξη είναι:

\textbf{Ευκλείδεια απόσταση} (L2): μετράει την ευθεία απόσταση μεταξύ δύο σημείων στον $\mathbb{R}^d$,
\[
d_{\text{L2}}(\mathbf{x},\mathbf{y})= \sqrt{\sum_{i=1}^{d}(x_i-y_i)^2}
\]
και χρησιμοποιείται ευρέως σε διανύσματα χαρακτηριστικών εικόνας και ήχου, όπου η γεωμετρική απόσταση στον χώρο αντιστοιχεί σε πραγματική ομοιότητα.

\textbf{Ομοιότητα Συνημιτόνου} (\selectlanguage{english}cosine similarity\selectlanguage{greek}): μετράει τη γωνία μεταξύ δύο διανυσμάτων ανεξάρτητα από το μήκος τους,
\[
\text{cosine\_sim}(\mathbf{x},\mathbf{y}) = \frac{\mathbf{x}\cdot\mathbf{y}}{\|\mathbf{x}\|\,\|\mathbf{y}\|}, \quad
d_{\text{cosine}}(\mathbf{x},\mathbf{y}) = 1 - \text{cosine\_sim}(\mathbf{x},\mathbf{y})
\]
Προτιμάται στα διανύσματα κειμένου (\selectlanguage{english}text embeddings\selectlanguage{greek}), όπου η κατεύθυνση στον χώρο κωδικοποιεί τη σημασιολογία, όχι το μέγεθος.

Η εξαντλητική αναζήτηση (\selectlanguage{english}brute-force\selectlanguage{greek}) --- υπολογισμός της απόστασης του ερωτήματος από κάθε ένα από τα $N$ στοιχεία --- εγγυάται βέλτιστα αποτελέσματα αλλά έχει κόστος $\mathcal{O}(N \cdot d)$ ανά ερώτημα. Για $N = 10^6$ και $d = 128$, αυτό αντιστοιχεί σε περίπου $1.28 \times 10^8$ πράξεις κινητής υποδιαστολής ανά ερώτημα --- αδύνατο να εκτελεστεί σε πραγματικό χρόνο σε παραγωγικά συστήματα, ανεξάρτητα από το υλικό. Η γραμμική κλιμάκωση ως προς το $N$ σημαίνει ότι η εξαντλητική αναζήτηση παύει να είναι εφαρμόσιμη καθώς ο αριθμός των δεδομένων μεγαλώνει.

\subsection{Η Κατάρα της Διαστατικότητας}

Ένα από τα θεμελιώδη εμπόδια στην αναζήτηση κοντινότερων γειτόνων σε υψηλές διαστάσεις είναι το φαινόμενο γνωστό ως \textbf{κατάρα της διαστατικότητας} (\selectlanguage{english}curse of dimensionality\selectlanguage{greek}). Το φαινόμενο αυτό περιγράφει τη συστηματική αποτυχία γεωμετρικών και αλγοριθμικών εργαλείων που λειτουργούν καλά σε χαμηλές διαστάσεις να κλιμακωθούν στις υψηλές.

Τυπικά, σε χώρο $d$ διαστάσεων, ο όγκος μιας σφαίρας ακτίνας $r$ αυξάνεται εκθετικά με $d$. Αυτό συνεπάγεται ότι, για να καλυφθεί σταθερό ποσοστό του χώρου με ένα πεπερασμένο σύνολο δεδομένων, χρειάζεται εκθετικά μεγαλύτερη πυκνότητα δεδομένων καθώς το $d$ αυξάνεται. Ένα άλλο σύμπτωμα: σε υψηλές διαστάσεις η διαφορά μεταξύ της ελάχιστης και της μέγιστης απόστασης ενός τυχαίου ερωτήματος από ένα σύνολο σημείων τείνει να μηδενίζεται σχετικά:
\[
\frac{d_{\max} - d_{\min}}{d_{\min}} \xrightarrow{d \to \infty} 0
\]
Αυτό σημαίνει ότι όλα τα σημεία τείνουν να είναι «εξίσου κοντά» στο ερώτημα, εκμηδενίζοντας την έννοια του «κοντινού γείτονα» και καθιστώντας τις δομές ευρετηρίου που βασίζονται σε διαμερισμό του χώρου (π.χ. \selectlanguage{english}KD-trees\selectlanguage{greek}) αναποτελεσματικές.

Η κατάρα της διαστατικότητας εξηγεί γιατί οι μέθοδοι που λειτουργούν άριστα για $d \leq 20$ (δενδρικές δομές, \selectlanguage{english}grid\selectlanguage{greek}-βασισμένη ευρετηρίαση) αδυνατούν να κλιμακωθούν σε $d = 128$ ή $d = 784$ --- και γιατί η ερευνητική κοινότητα στράφηκε σε αλγόριθμους που λειτουργούν σε επίπεδο γράφου ή κατακερματισμού, παρακάμπτοντας την ευκλείδεια γεωμετρία.

\subsection{Μέθοδοι Κατά Προσέγγιση Αναζήτησης Κοντινότερων Γειτόνων}

Δεδομένου ότι η εξαντλητική αναζήτηση δεν κλιμακώνεται, η ερευνητική κοινότητα έχει αναπτύξει μεθόδους \textbf{κατά προσέγγιση αναζήτησης κοντινότερων γειτόνων} (\selectlanguage{english}Approximate Nearest Neighbor -- ANN\selectlanguage{greek}) \cite{nns_survey}. Αυτές οι μέθοδοι θυσιάζουν ελεγχόμενο ποσοστό ακρίβειας --- δηλαδή δεν εγγυώνται ότι θα επιστρέψουν πάντα τους ακριβώς βέλτιστους $k$ γείτονες --- έναντι σημαντικής επιτάχυνσης. Στην πράξη, \selectlanguage{english}recall\selectlanguage{greek} της τάξης $0.95$--$0.99$ είναι αρκετό για τις περισσότερες εφαρμογές, και επιτυγχάνεται με πολλαπλάσια ταχύτητα σε σχέση με τη βέλτιστη λύση.

Υπάρχουν τρεις κύριες οικογένειες μεθόδων:

\subsubsection{Μέθοδοι βασισμένες σε Κατακερματισμό}

Οι μέθοδοι κατακερματισμού (\selectlanguage{english}Hashing\selectlanguage{greek}) ήταν ανάμεσα στις πρώτες προτάσεις για ANN. Η πιο γνωστή είναι το \selectlanguage{english}\textbf{Locality-Sensitive Hashing (LSH)}\selectlanguage{greek}: χρησιμοποιεί οικογένειες τυχαίων συναρτήσεων κατακερματισμού με την ιδιότητα να τοποθετούν κοντινά διανύσματα στον ίδιο κάδο με υψηλή πιθανότητα. Κατά την αναζήτηση, εξετάζονται μόνο τα στοιχεία των κάδων στους οποίους αντιστοιχεί το ερώτημα, αποφεύγοντας τους υπόλοιπους.

Για ευκλείδεια απόσταση, μια χαρακτηριστική \selectlanguage{english}LSH\selectlanguage{greek} συνάρτηση ορίζεται ως:
\[
h_{\mathbf{a}, b}(\mathbf{x}) = \left\lfloor \frac{\mathbf{a} \cdot \mathbf{x} + b}{w} \right\rfloor
\]
όπου $\mathbf{a} \in \mathbb{R}^d$ τυχαίο διάνυσμα από κανονική κατανομή, $b \sim \text{Uniform}(0, w)$, και $w > 0$ το πλάτος του κάδου. Η ιδιότητα που κάνει αυτές τις συναρτήσεις χρήσιμες:
\[
\Pr[h(\mathbf{x}) = h(\mathbf{y})] = 1 - \frac{d(\mathbf{x},\mathbf{y})}{w} + O\!\left(\frac{d(\mathbf{x},\mathbf{y})^2}{w^2}\right)
\]
δηλαδή, σημεία κοντά στον χώρο ($d(\mathbf{x},\mathbf{y})$ μικρό) έχουν υψηλή πιθανότητα να πέσουν στον ίδιο κάδο. Στην πράξη, χρησιμοποιούνται $L$ ανεξάρτητες ομάδες από $K$ τέτοιες συναρτήσεις για να αυξηθεί η ακρίβεια, με κόστος πολλαπλής χρήσης μνήμης.

\textbf{Πλεονεκτήματα:} απλή υλοποίηση, θεωρητικές εγγυήσεις ακρίβειας, χαμηλό κόστος κατασκευής ευρετηρίου.

\textbf{Μειονεκτήματα:} σε υψηλές διαστάσεις ($d > 100$) η αποδοτικότητα μειώνεται σημαντικά, καθώς η τοπική ευαισθησία των συναρτήσεων εξασθενεί --- απαιτούνται πολλαπλές ανεξάρτητες ομάδες κάδων για αποδεκτό \selectlanguage{english}recall\selectlanguage{greek}, με αντίστοιχη αύξηση μνήμης και χρόνου αναζήτησης. Στην πράξη, για υψηλές διαστάσεις τείνουν να υπολείπονται των γραφο-βασισμένων μεθόδων \cite{nns_survey}.

\subsubsection{Μέθοδοι βασισμένες σε Κβαντοποίηση}

Οι μέθοδοι κβαντοποίησης (\selectlanguage{english}Quantization\selectlanguage{greek}) ακολουθούν διαφορετική φιλοσοφία: αντί να επιταχύνουν τη διαδικασία αναζήτησης αλλάζοντας τον αλγόριθμο, συμπιέζουν τα ίδια τα διανύσματα σε συμπαγείς αναπαραστάσεις. Η πιο διαδεδομένη μέθοδος είναι το \selectlanguage{english}\textbf{Product Quantization (PQ)}\selectlanguage{greek}: το κάθε διάνυσμα $d$ διαστάσεων χωρίζεται σε $m$ υποδιανύσματα ίσου μεγέθους $d/m$, και το καθένα κβαντοποιείται ανεξάρτητα χρησιμοποιώντας ένα λεξιλόγιο κωδικών (\selectlanguage{english}codebook\selectlanguage{greek}). Κατά την αναζήτηση, οι αποστάσεις υπολογίζονται απευθείας από τα κωδικολόγια, χωρίς να αποκωδικοποιηθούν τα πλήρη διανύσματα.

\textbf{Παράδειγμα:} Έστω $d = 4$ και $m = 2$ υποδιανύσματα. Ένα διάνυσμα $\mathbf{x} = (x_1, x_2, x_3, x_4)$ διασπάται σε $\mathbf{x}^{(1)} = (x_1, x_2)$ και $\mathbf{x}^{(2)} = (x_3, x_4)$. Το $k$-\selectlanguage{english}means\selectlanguage{greek} εκτελείται ανεξάρτητα σε κάθε υποχώρο παράγοντας 256 centroids ανά υποχώρο (1 byte ανά υπο-κωδικό). Τελικά, κάθε διάνυσμα αναπαρίσταται από 2 bytes αντί $4 \times 4 = 16$ bytes (σε \selectlanguage{english}float32\selectlanguage{greek}), με παράγοντα συμπίεσης $8\times$. Κατά τη φάση αναζήτησης, οι αποστάσεις ανάμεσα στο ερώτημα και κάθε \selectlanguage{english}centroid\selectlanguage{greek} προϋπολογίζονται (\selectlanguage{english}lookup tables\selectlanguage{greek}) και η απόσταση $\hat{d}(\mathbf{q}, \mathbf{x})$ υπολογίζεται ως άθροισμα τιμών πίνακα:
\[
\hat{d}(\mathbf{q}, \mathbf{x}) \approx \sum_{j=1}^{m} d\!\left(\mathbf{q}^{(j)},\, c^{(j)}_{I_j(\mathbf{x})}\right)
\]
όπου $I_j(\mathbf{x})$ ο κωδικός του $j$-ου υποδιανύσματος και $c^{(j)}_{I_j}$ ο αντίστοιχος \selectlanguage{english}centroid\selectlanguage{greek}.

\textbf{Πλεονεκτήματα:} εξαιρετικά χαμηλή κατανάλωση μνήμης (4--32 bytes ανά διάνυσμα ανεξαρτήτως $d$) και υψηλή απόδοση σε συστήματα με σφιχτούς μνημονικούς περιορισμούς. Συνδυάζεται συχνά με \selectlanguage{english}Inverted File Index (IVF)\selectlanguage{greek} για αναζήτηση δισεκατομμυρίων διανυσμάτων, όπως στην υλοποίηση \selectlanguage{english}FAISS \cite{faiss2017}\selectlanguage{greek}.

\textbf{Μειονεκτήματα:} η συμπίεση εισάγει αναπόφευκτο σφάλμα κβαντοποίησης --- οι εκτιμήσεις αποστάσεων δεν είναι ακριβείς, και το \selectlanguage{english}recall\selectlanguage{greek} μειώνεται όσο αυξάνεται η συμπίεση. Για υψηλές απαιτήσεις \selectlanguage{english}recall\selectlanguage{greek} ($>0.95$) σε υψηλές διαστάσεις, η κβαντοποίηση μόνη της δεν επαρκεί.

\subsubsection{Μέθοδοι βασισμένες σε Γράφους}

Στις μεθόδους αυτές, τα δεδομένα οργανώνονται σε ένα γράφο $G = (V, E)$: κάθε κόμβος $v \in V$ αντιστοιχεί σε ένα διάνυσμα της συλλογής, και κάθε ακμή $(u, v) \in E$ συνδέει δύο κόμβους που είναι γείτονες (κοντινοί) στον μετρικό χώρο. Κατά την αναζήτηση, ο αλγόριθμος πλοηγείται στον γράφο: ξεκινά από έναν κόμβο εισόδου και σε κάθε βήμα μεταβαίνει στον γειτονικό κόμβο που μειώνει περισσότερο την απόσταση από το ερώτημα, έως ότου δεν βρεθεί καλύτερος γείτονας.

Η αποδοτικότητα εξαρτάται κρίσιμα από τη δομή των ακμών: ένα καλό γράφο ANN έχει ακμές τόσο τοπικής εμβέλειας (που εξασφαλίζουν ακρίβεια στη λεπτομερή αναζήτηση) όσο και μακράς εμβέλειας («γέφυρες» που επιτρέπουν γρήγορη μετακίνηση σε διαφορετικές περιοχές του χώρου). Αυτή η ιδιότητα ονομάζεται \textbf{δομή μικρού κόσμου} (\selectlanguage{english}navigable small world\selectlanguage{greek}) και εξηγεί γιατί η αναζήτηση με πλοήγηση βρίσκει τους κοντινούς γείτονες σε λογαριθμικό αριθμό βημάτων \cite{malkov2018hnsw}.

\textbf{Πλεονεκτήματα:} υψηλό \selectlanguage{english}recall\selectlanguage{greek} για δεδομένο αριθμό υπολογισμών απόστασης, ευελιξία ως προς το μέτρο απόστασης, και δυνατότητα δυναμικής εισαγωγής νέων δεδομένων χωρίς ανακατασκευή. Κυριαρχούν στα σύγχρονα \selectlanguage{english}benchmarks\selectlanguage{greek} για υψηλές διαστάσεις.

\textbf{Μειονεκτήματα:} υψηλότερη κατανάλωση μνήμης λόγω αποθήκευσης ακμών, και μεγαλύτερο κόστος κατασκευής ευρετηρίου σε σχέση με τις υπόλοιπες οικογένειες. Η απόδοση εξαρτάται από την ποιότητα του κατασκευασμένου γράφου.

Το \selectlanguage{english}\textbf{HNSW}\selectlanguage{greek} αποτελεί την πιο ευρέως υιοθετημένη υλοποίηση αυτής της κατηγορίας και αναλύεται εκτενώς στην επόμενη ενότητα.

\subsection{Υπολογιστικό Κόστος Αναζήτησης}
Στην εξαντλητική αναζήτηση, το ερώτημα συγκρίνεται με όλα τα $Ν$ διανύσματα της βάσης δεδομένων διάστασης \selectlanguage{english}$d$\selectlanguage{greek}, οδηγώντας σε υπολογιστικό κόστος της τάξης:
\[
\mathcal{O}(N\cdot d)
\]
Σε εφαρμογές μεγάλης κλίμακας, όπου $N$ μπορεί να μετράει εκατομμύρια ή ακόμα και δισεκατομμύρια διανύσματα, το κόστος κυριαρχείται από τον αριθμό υπολογισμών απόστασης. Αυτός είναι και ο λόγος που η αποδοτικότητα ενός αλγορίθμου ANN δεν κρίνεται μόνο από τον χρόνο εκτέλεσης, αλλά --- και ίσως κυρίως --- από τον αριθμό υπολογισμών απόστασης που απαιτούνται ανά ερώτημα. Ο χρόνος είναι εξαρτώμενος από το υλικό· οι υπολογισμοί απόστασης δεν είναι.

\subsection{\selectlanguage{greek}Hierarchical Navigable Small World Graphs (HNSW)}
Ο αλγόριθμος \selectlanguage{english}\textbf{Hierarchical Navigable Small World (HNSW)}\selectlanguage{greek} ανήκει στην κατηγορία των \selectlanguage{english}graph-based ANN\selectlanguage{greek} μεθόδων και αποτελεί σήμερα την κυρίαρχη επιλογή σε παραγωγικά συστήματα αναζήτησης ομοιότητας, με υλοποιήσεις στις περισσότερες βιβλιοθήκες ανοιχτού κώδικα. Η βασική του αρχή βασίζεται στη διαστρωμάτωση των δεδομένων σε πολλαπλά επίπεδα γράφων μικρού κόσμου.

Η αναζήτηση εκκινεί από τα ανώτερα, αραιά επίπεδα --- όπου κάθε κόμβος έχει λίγους αλλά μακράς εμβέλειας γείτονες --- και κατεβαίνει σταδιακά προς το πυκνό κατώτερο επίπεδο (επίπεδο 0), όπου εντοπίζονται οι τελικοί γείτονες. Τα ανώτερα επίπεδα επιτελούν ρόλο πλοήγησης υψηλού επιπέδου: οδηγούν γρήγορα προς τη σωστή περιοχή του χώρου, χωρίς να εξαντλούν το υπολογιστικό budget. Η λεπτομερής αναζήτηση γίνεται αποκλειστικά στο επίπεδο 0.

Η αποτελεσματικότητα αυτής της δομής οφείλεται στην αρχή της ιεραρχικής εξερεύνησης: η πληροφορία οργανώνεται από το γενικό στο ειδικό, επιτρέποντας γρήγορη σύγκλιση με περιορισμένο αριθμό υπολογισμών απόστασης. Ο συνδυασμός αυτός καθιστά τον αλγόριθμο ιδιαίτερα αποδοτικό σε μεγάλα σύνολα δεδομένων υψηλής διάστασης --- όπου άλλες προσεγγίσεις υπολείπονται.

\subsubsection{Δομή του Πολυεπίπεδου Γράφου}
Στο επίπεδο δομής, ο \selectlanguage{english}HNSW\selectlanguage{greek} είναι μια στοίβα από γράφους μικρού κόσμου --- ο ένας πάνω στον άλλον. Κάθε κόμβος αντιστοιχεί σε ένα διάνυσμα και μπορεί να εμφανιστεί σε ένα ή περισσότερα επίπεδα, ανάλογα με τον τυχαίο αριθμό που του ανατέθηκε κατά την εισαγωγή.

Τα ανώτερα επίπεδα είναι αραιά: λίγοι κόμβοι, λίγες ακμές, αλλά μεγάλη εμβέλεια. Στο κατώτερο επίπεδο (επίπεδο 0) βρίσκεται το πλήρες σύνολο δεδομένων με πυκνές συνδέσεις --- εκεί εκτελείται η λεπτομερής τελική αναζήτηση.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.3\textwidth]{images/hnsw_layers.jpg}
    \caption{
    Τα ανώτερα επίπεδα περιέχουν λιγότερους κόμβους, ενώ το επίπεδο 0 περιλαμβάνει όλα τα δεδομένα.}
\label{fig:hnsw_structure}
\end{figure}


Η ανάθεση κόμβων στα επίπεδα γίνεται τυχαία, ακολουθώντας γεωμετρική κατανομή. Στην πράξη αυτό σημαίνει ότι η συντριπτική πλειοψηφία των κόμβων εντάσσεται μόνο στο επίπεδο 0, λίγοι φτάνουν στο επίπεδο 1 ή 2, και μόνο ελάχιστοι αντιστοιχίζονται σε ανώτερα επίπεδα. Αποτέλεσμα της επιλογής αυτής της κατανομής είναι ότι τα ανώτερα επίπεδα παραμένουν αραιά, με συνέπεια η πλοήγηση εκεί να είναι ταχεία και να μην απαιτείται εκτεταμένη διερεύνηση γειτόνων.

Κάθε κόμβος συνδέεται με περιορισμένο αριθμό γειτόνων σε κάθε επίπεδο που εμφανίζεται --- αυτό το όριο ελέγχεται από την παράμετρο $M$. Η πλοήγηση ξεκινά από τα ανώτερα επίπεδα και κατεβαίνει σταδιακά, με κάθε επίπεδο να επικεντρώνει την αναζήτηση σε μια όλο και πιο συγκεκριμένη περιοχή του χώρου.

\subsubsection{Παράδειγμα Λειτουργίας}
\label{sec:hnsw_example}

Για να γίνει σαφής η λειτουργία του αλγορίθμου, εξετάζουμε ένα απλοποιημένο παράδειγμα με 7 κόμβους σε χώρο δύο διαστάσεων. Έστω τα εξής σημεία:

\[
A=(1,8),\; B=(3,7),\; C=(5,5),\; D=(7,6),\; E=(2,3),\; F=(6,2),\; G=(8,4)
\]

Μετά την εισαγωγή όλων των κόμβων, υποθέτουμε την εξής κατανομή επιπέδων (αποτέλεσμα τυχαίας δειγματοληψίας):
\begin{itemize}
    \item Επίπεδο 2: $\{A\}$
    \item Επίπεδο 1: $\{A, C\}$
    \item Επίπεδο 0: $\{A, B, C, D, E, F, G\}$ (όλοι)
\end{itemize}

Έστω ερώτημα $q = (6, 3)$ και $k=1$, $ef_{\text{search}}=2$.

\textbf{Φάση 1 --- Greedy στα ανώτερα επίπεδα:}
\begin{itemize}
    \item \textit{Επίπεδο 2:} Εκκίνηση από $A$. Μόνος κόμβος εδώ, οπότε $ep \leftarrow A$.
    \item \textit{Επίπεδο 1:} Από $A\,(d=7.07)$, εξετάζεται ο μοναδικός γείτονας $C\,(d=2.24)$. Επειδή $d(q,C) < d(q,A)$, $ep \leftarrow C$.
\end{itemize}

\textbf{Φάση 2 --- Best-first στο επίπεδο 0:}
Αρχή από $ep = C\,(d=2.24)$. Αρχικοποίηση: υποψήφιοι $= \{C\}$, αποτελέσματα $= \{C\}$.
\begin{enumerate}
    \item Επεκτείνεται ο $C$. Γείτονές του στο επίπεδο 0: $\{B, D, F\}$. Υπολογίζονται αποστάσεις: $d(q,B)=5.00$, $d(q,D)=2.83$, $d(q,F)=1.00$. Ο $F$ είναι ο πιο κοντινός.
    \item Ενημέρωση: αποτελέσματα $= \{F, C\}$ (τα 2 καλύτερα).
    \item Επεκτείνεται ο $F\,(d=1.00)$. Γείτονές του: $\{C, G, E\}$. $d(q,G)=2.83$, $d(q,E)=4.12$. Δεν βελτιώνουν τα τρέχοντα αποτελέσματα.
    \item Τερματισμός: ο επόμενος υποψήφιος ($D$, $d=2.83$) είναι χειρότερος από τον $F$ στα αποτελέσματα.
\end{enumerate}

\textbf{Αποτέλεσμα:} $F=(6,2)$, που πράγματι είναι ο κοντινότερος γείτονας του $q=(6,3)$ ($d=1.00$). Η αναζήτηση εκτέλεσε μόνο 6 υπολογισμούς απόστασης αντί για 7 (εξαντλητική), αποφεύγοντας τους απομακρυσμένους κόμβους $A$ και $E$.


\subsubsection{Διαδικασία Εισαγωγής Νέου Κόμβου}
Για κάθε νέο κόμβο επιλέγεται τυχαία το ανώτερο επίπεδο στο οποίο θα εμφανιστεί. Ο τρόπος δειγματοληψίας έχει επιλεγεί ώστε οι περισσότεροι κόμβοι να τοποθετούνται μόνο στα χαμηλά επίπεδα, ενώ λίγοι να εμφανίζονται και στα ανώτερα --- έτσι δημιουργείται η ιεράρχηση που κάνει τον αλγόριθμο να λειτουργεί.

Ακολουθεί φάση πλοήγησης: εκκινώντας από το ανώτερο επίπεδο, εκτελείται \selectlanguage{english}greedy\selectlanguage{greek} αναζήτηση προς τον εντοπισμό της κατάλληλης περιοχής για σύνδεση του νέου κόμβου. Η διαδικασία επαναλαμβάνεται καθοδικά ανά επίπεδο, έως ότου η αναζήτηση φτάσει στο επίπεδο βάσης.

Σε κάθε επίπεδο, ο νέος κόμβος συνδέεται με περιορισμένο αριθμό γειτονικών κόμβων βάσει ευρετικών κανόνων επιλογής. Οι κανόνες αυτοί δεν επιλέγουν αποκλειστικά τους πλησιέστερους κόμβους --- εξασφαλίζουν παράλληλα ποικιλομορφία ακμών, ώστε ο γράφος να παραμένει αραιός και η αναζήτηση σε αυτόν αποδοτική.

Συνοπτικά, η εισαγωγή ακολουθεί αυτή τη σειρά:
\begin{itemize}
    \item Τυχαία επιλογή του ανώτερου επιπέδου εμφάνισης — η γεωμετρική κατανομή εξασφαλίζει τη σταδιακή αραίωση των ανώτερων επιπέδων.
    \item Εκκίνηση από το ανώτερο επίπεδο του γράφου και \selectlanguage{english}greedy\selectlanguage{greek} αναζήτηση καθοδικά, ανά επίπεδο.
    \item Στο επίπεδο εισαγωγής και κάτω: εφαρμογή best-first search (πιο εκτενής εξερεύνηση) για καλύτερη ποιότητα γράφου.
    \item Επιλογή και σύνδεση με έως $M$ γείτονες σε κάθε επίπεδο — όχι απαραίτητα τους $M$ πιο κοντινούς, αλλά αυτούς που εξασφαλίζουν ποικιλομορφία ακμών.
\end{itemize}

\subsubsection{Παράδειγμα Εισαγωγής και Επιλογής Γειτόνων}

Χρησιμοποιούμε τον ίδιο γράφο 7 κόμβων του §\ref{sec:hnsw_example} και εισάγουμε έναν νέο κόμβο $H = (4, 3)$ με παραμέτρους $M = 2$ και $ef_{\text{construction}} = 3$.

\textbf{Βήμα 1 — Ανάθεση επιπέδου.} Η γεωμετρική κατανομή αναθέτει τυχαία επίπεδο στο $H$. Έστω ότι $H$ ανατίθεται στο επίπεδο 0 (συχνότερη περίπτωση).

\textbf{Βήμα 2 — Greedy πλοήγηση.} Εκκίνηση από τον κόμβο εισόδου $A = (1,8)$ στο επίπεδο 2.
\begin{itemize}
    \item \textit{Επίπεδο 2:} Μόνος κόμβος ο $A\,(d=4.24)$, οπότε $ep \leftarrow A$.
    \item \textit{Επίπεδο 1:} Από $A$, εξετάζεται ο γείτονας $C = (5,5)\,(d=2.24)$. Επειδή $d(H,C) < d(H,A)$, $ep \leftarrow C$.
\end{itemize}

\textbf{Βήμα 3 — Best-first search στο επίπεδο 0.} Εκκίνηση από $ep = C$ με $ef_{\text{construction}} = 3$. Αποτελέσματα που επιστρέφει η αναζήτηση:

\[
d(H,E) = 2.00,\quad d(H,C) = 2.24,\quad d(H,F) = 2.24
\]

Οι 3 καλύτεροι υποψήφιοι: $\mathcal{C} = \{E, C, F\}$.

\textbf{Βήμα 4 — SelectNeighbors.} Ο $H$ ανήκει στο επίπεδο 0, οπότε το ανώτατο όριο γειτόνων είναι $M_0 = 2M = 4$ (όχι $M = 2$). Η ευρετική επιλογή δεν επιλέγει απλώς τους κοντινότερους· ελέγχει ποικιλομορφία (στην πράξη επιλέγονται 2 γείτονες λόγω του κριτηρίου ποικιλομορφίας, όχι λόγω του ορίου $M_0$):

\begin{enumerate}
    \item Προσθέτουμε $E$ (πλησιέστερος, $d=2.00$) στο σύνολο αποτελεσμάτων $R \leftarrow \{E\}$.
    \item Εξετάζουμε $C\,(d=2.24)$: ελέγχουμε αν ο $C$ «κρύβεται» πίσω από τον $E$.
        \[ d(C, E) = \sqrt{(5{-}2)^2 + (5{-}3)^2} = \sqrt{13} \approx 3.61 \]
        Επειδή $d(C,E) = 3.61 > d(C,H) = 2.24$, ο $C$ \emph{δεν} κρύβεται: προσθέτουμε $C$ στο $R$.
    \item $|R| = 2 = M$ — σταματάμε. Ο $F$ απορρίπτεται.
\end{enumerate}

\textbf{Βήμα 5 — Αμφίδρομες ακμές.} Δημιουργούνται οι συνδέσεις $H \leftrightarrow E$ και $H \leftrightarrow C$ στο επίπεδο 0. Αν ο $E$ ή ο $C$ υπερβαίνει πλέον το $M_0 = 4$ γειτόνων (το όριο του επιπέδου 0), εφαρμόζεται εκ νέου η SelectNeighbors στη λίστα γειτόνων τους για να συρρικνωθεί.

\textbf{Σχόλιο για διαγραφές.} Ο \selectlanguage{english}HNSW\selectlanguage{greek} δεν υποστηρίζει εγγενή διαγραφή κόμβων. Δύο προσεγγίσεις αντιμετωπίζουν αυτόν τον περιορισμό: (α) \textit{soft deletion} — ο κόμβος σημαίνεται ως διαγραμμένος και παρακάμπτεται κατά την αναζήτηση, χωρίς τροποποίηση του γράφου, και (β) \textit{τοπική επαναδόμηση} — αφαιρούνται οι ακμές του διαγραμμένου κόμβου και οι γείτονές του επανασυνδέονται. Η πρώτη προσέγγιση είναι απλούστερη αλλά υποβαθμίζει σταδιακά την ποιότητα του γράφου· η δεύτερη είναι πιο δαπανηρή υπολογιστικά.

\paragraph{\selectlanguage{english}\texttt{Greedy Search:}\selectlanguage{greek}}
Στα ανώτερα επίπεδα ο αλγόριθμος ακολουθεί greedy στρατηγική: ξεκινά από έναν κόμβο εισόδου και σε κάθε βήμα μεταβαίνει στον γείτονα που μειώνει περισσότερο την απόσταση από το ερώτημα. Η επιλογή της \selectlanguage{english}C++\selectlanguage{greek} έγινε για λόγους απόδοσης, ενώ η γέφυρα προς \selectlanguage{english}Python\selectlanguage{greek} διευκολύνει τη χρήση της σε πειραματικά σενάρια. Ο κώδικας οργανώνεται σε τρία διακριτά τμήματα: τις δομές δεδομένων του γράφου, τη διαδικασία εισαγωγής κόμβων και τη διαδικασία αναζήτησης.


\subsection{Δομές Δεδομένων}
Υπάρχουν κάποιες επιλογές σχεδιασμού που αξίζει να αναφερθούν εδώ, γιατί επηρεάζουν άμεσα τις προσαρμοστικές παραλλαγές που ακολουθούν. Οι βασικές δομές:
\begin{itemize}
    \item \textbf{Αποθήκευση διανυσμάτων:} συνεχόμενος πίνακας \selectlanguage{english}$N \times d$\selectlanguage{greek} (\selectlanguage{english}row-major\selectlanguage{greek}), όπου κάθε γραμμή αντιστοιχεί σε ένα διάνυσμα του συνόλου δεδομένων. Η συνεχόμενη διάταξη βελτιώνει την τοπικότητα \selectlanguage{english}cache\selectlanguage{greek} κατά τον υπολογισμό αποστάσεων. \textit{Χώρος: $O(N \cdot d)$.}
    \item \textbf{Επίπεδα κόμβων:} πίνακας $N$ ακεραίων, όπου η θέση $u$ αποθηκεύει το μέγιστο επίπεδο $L_u$ στο οποίο εμφανίζεται ο κόμβος $u$. Χρησιμοποιείται κατά την εισαγωγή για να περιοριστεί η κατανομή ακμών ανά επίπεδο. \textit{Χώρος: $O(N)$.}
    \item \textbf{Λίστες γειτόνων ανά επίπεδο:} για κάθε κόμβο $u$ και κάθε επίπεδο $l \le L_u$, μία λίστα από \selectlanguage{english}node ids\selectlanguage{greek} γειτόνων. Στο επίπεδο 0 κάθε κόμβος έχει έως $2M$ γείτονες· στα ανώτερα επίπεδα έως $M$. Αυτή η δομή αποτελεί τον κύριο παράγοντα χρήσης μνήμης. \textit{Χώρος: $O(N \cdot M)$ συνολικά σε όλα τα επίπεδα.}
    \item \selectlanguage{english}\textbf{Visited}\selectlanguage{greek} \textbf{(σύνολο επισκεφθέντων κόμβων):} πίνακας $N$ χρονοσφραγίδων (\selectlanguage{english}timestamp array\selectlanguage{greek}) --- σε κάθε θέση $u$ αποθηκεύεται το αναγνωριστικό του τελευταίου ερωτήματος που επισκέφθηκε τον $u$. Αντί να μηδενίζεται ο πίνακας μεταξύ ερωτημάτων, αυξάνεται απλώς ο μετρητής ερωτημάτων· ο έλεγχος $\text{visited}[u] = \text{query\_id}$ αποτρέπει επανεπίσκεψη με $O(1)$ ανά κόμβο. \textit{Χώρος: $O(N)$.}
    \item \textbf{Ουρές προτεραιότητας:} δύο \selectlanguage{english}binary heaps\selectlanguage{greek} ανά ερώτημα --- μία \selectlanguage{english}min-heap\selectlanguage{greek} για τους υποψηφίους προς επέκταση (ταξινομημένους κατ' αύξουσα απόσταση από το ερώτημα) και μία \selectlanguage{english}max-heap\selectlanguage{greek} για το τρέχον σύνολο αποτελεσμάτων μεγέθους $ef_{\text{search}}$ (ώστε ο κόμβος με τη μεγαλύτερη απόσταση να αφαιρείται εύκολα). \textit{Χώρος: $O(ef_{\text{search}})$ ανά ερώτημα.}
\end{itemize}
Ο διαχωρισμός \selectlanguage{english}candidate queue\selectlanguage{greek} και \selectlanguage{english}top set\selectlanguage{greek} αποτελεί συνειδητή αρχιτεκτονική επιλογή --- η διάκριση αυτή επιτρέπει τον έλεγχο του \selectlanguage{english}search effort\selectlanguage{greek} μέσω του \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek}, και συνιστά τη βάση πάνω στην οποία εδράζονται οι προσαρμοστικές παραλλαγές του επόμενου κεφαλαίου.

\subsubsection{Ανάθεση επιπέδου (\selectlanguage{english}Layer Assignment\selectlanguage{greek})}
Κάθε νέος κόμβος $q$ αποκτά τυχαία ένα μέγιστο επίπεδο $L$, ακολουθώντας εκθετική κατανομή· ο κόμβος εισάγεται κατόπιν σε όλα τα επίπεδα $0, 1, \ldots, L$, ώστε τα ανώτερα επίπεδα να παραμένουν αραιά:
\begin{equation}
    P(L \ge l) = e^{-l/\lambda}
\end{equation}
όπου $\lambda$ είναι παράμετρος κλίμακας που συνδέεται με το $M$ μέσω της σχέσης:
\begin{equation}
    \lambda = \frac{1}{\ln{M}}
\end{equation}
Έτσι, η πλειοψηφία των κόμβων εμφανίζεται μόνο στο επίπεδο 0, ενώ ελάχιστοι φτάνουν στα ανώτερα επίπεδα --- η ιδιότητα που εξασφαλίζει τη γρήγορη πλοήγηση.

\subsubsection{\selectlanguage{english}Greedy\selectlanguage{greek} πλοήγηση στα ανώτερα επίπεδα}
Ο αλγόριθμος ξεκινά από έναν μοναδικό κόμβο εισόδου $c$, ο οποίος αποθηκεύεται ως καθολική μεταβλητή της δομής και είναι ο κόμβος που εισήχθη με το μεγαλύτερο επίπεδο μέχρι εκείνη τη στιγμή (ανανεώνεται όταν κάποιος νέος κόμβος αποκτά ακόμα υψηλότερο επίπεδο). Εκτελείται \selectlanguage{english}greedy search\selectlanguage{greek} σε κάθε επίπεδο $l$ από το ανώτερο προς τα κάτω, με τον $c$ να ανανεώνεται στον πλησιέστερο κόμβο που βρέθηκε πριν κατεβούμε στο επόμενο επίπεδο:
\begin{equation}
    c \leftarrow \arg\min_{v \in \mathcal{N}_l(c)} d(q,v),
\end{equation}
όπου $\mathcal{N}_l(c)$ είναι οι γείτονες του $c$ στο επίπεδο $l$ και $d(\cdot,\cdot)$ η μετρική απόσταση. Η διαδικασία αυτή, με έναν μόνο υποψήφιο ανά επίπεδο, εκτελείται για όλα τα επίπεδα $l > L$, ώστε ο αλγόριθμος να φτάσει στο επίπεδο εισαγωγής $L$ με έναν αξιόπιστο αρχικό κόμβο

\subsubsection{Αναζήτηση υποψηφίων στο επίπεδο βάσης (\selectlanguage{english}level 0\selectlanguage{greek})}
Στο επίπεδο βάσης η \selectlanguage{english}greedy search\selectlanguage{greek} με ένα μόνο υποψήφιο δεν επαρκεί --- χρειάζεται καλύτερη τοπική διερεύνηση για να παραχθεί ποιοτικός γράφος. Εφαρμόζεται αναζήτηση τύπου \selectlanguage{english}best-first search\selectlanguage{greek}, κατά την οποία η παράμετρος \selectlanguage{english}$ef_{\text{construction}}$\selectlanguage{greek} ορίζει το μέγεθος της δυναμικής λίστας υποψηφίων: πόσους κόμβους κρατάμε ενεργούς κατά την εξερεύνηση. Από αυτούς τους υποψηφίους επιλέγονται στη συνέχεια το πολύ $M$ τελικοί γείτονες για σύνδεση με τον νέο κόμβο --- ο $M$ δηλαδή καθορίζει τον αριθμό γειτόνων, ενώ το \selectlanguage{english}$ef_{\text{construction}}$\selectlanguage{greek} καθορίζει την ποιότητα της υποψήφιας δεξαμενής από την οποία επιλέγονται. Υψηλότερη τιμή \selectlanguage{english}$ef_{\text{construction}}$\selectlanguage{greek} σημαίνει καλύτερες ακμές στον γράφο --- αλλά και μεγαλύτερο κόστος κατασκευής.


\subsubsection{Επιλογή γειτόνων και δημιουργία ακμών}
Από το σύνολο των υποψηφίων κόμβων στο επίπεδο $l$ επιλέγονται έως $M$ γείτονες για σύνδεση με τον νέο κόμβο $q$. Εξαίρεση αποτελεί το επίπεδο βάσης ($l = 0$), όπου το όριο είναι $M_0 = 2M$ --- η αύξηση αυτή βελτιώνει τη συνδεσιμότητα του πυκνού κατώτερου επιπέδου, όπου οι κόμβοι συγκεντρώνουν το μεγαλύτερο υπολογιστικό φόρτο κατά την αναζήτηση. Η επιλογή δεν βασίζεται αποκλειστικά στην ελάχιστη απόσταση από το $q$ --- εφαρμόζεται ευρετική που ενισχύει την ποικιλομορφία των ακμών και αποτρέπει πλεονάζουσες συνδέσεις που θα επιβάρυναν τη μνήμη χωρίς να βελτιώνουν την ποιότητα αναζήτησης.

Ένα τυπικό κριτήριο αποδοχής ενός υποψηφίου $u$ στο σύνολο επιλεγμένων γειτόνων $S$ είναι:
\begin{equation}
    \forall w \in S:\quad d(u,q) < d(u,w)
\end{equation}
δηλαδή ο $u$ γίνεται δεκτός μόνο αν δεν ``καλύπτεται'' από κάποιον ήδη επιλεγμένο γείτονα $w$.
Τέλος, οι ακμές εισάγονται αμφίδρομα (\selectlanguage{english}bidirectional links\selectlanguage{greek}) μεταξύ $q$ και των επιλεγμένων γειτόνων. Επειδή η νέα αμφίδρομη ακμή προσθέτει τον $q$ στη λίστα γειτόνων των υπαρχόντων κόμβων, κάποιοι από αυτούς ενδέχεται να ξεπεράσουν το επιτρεπτό όριο $M$. Στην περίπτωση αυτή εφαρμόζεται τοπικό \selectlanguage{english}pruning\selectlanguage{greek}: εκτελείται εκ νέου η ίδια ευρετική επιλογής γειτόνων στο διευρυμένο σύνολο γειτόνων του κόμβου, και κρατούνται μόνο οι $M$ που πληρούν το κριτήριο ποικιλομορφίας (δηλαδή δεν «καλύπτονται» από κάποιον ήδη επιλεγμένο γείτονα).
\medskip
Η παραπάνω διαδικασία επαναλαμβάνεται για όλα τα επίπεδα  $l \le L$ στα οποία ανήκει ο κόμβος $q$, επιτυγχάνοντας σταδιακή ενσωμάτωση του νέου κόμβου στον ιεραρχικό γράφο.

\subsection{Αναζήτηση \selectlanguage{english}(Querying)}
Η διαδικασία αναζήτησης στο \selectlanguage{english}HNSW\selectlanguage{greek} στοχεύει στην εύρεση των $k$ πλησιέστερων γειτόνων ενός ερωτήματος $q$, αξιοποιώντας την ιεραρχική δομή του γράφου για αποδοτικό προσεγγιστικό εντοπισμό. Η ταχύτητα εξαρτάται από το πόσα επίπεδα υπάρχουν και από τη σύνδεση του γράφου στο επίπεδο 0.

Η αναζήτηση εκτελείται σε δύο διακριτά στάδια:
\begin{itemize}
    \item πλοήγηση στα ανώτερα επίπεδα του γράφου και 
    \item λεπτομερής αναζήτηση στο επίπεδο βάσης
\end{itemize}

\subsubsection{Πλοήγηση στα ανώτερα επίπεδα}
Η διαδικασία ξεκινά από έναν κόμβο εισόδου $c$ στο ανώτερο επίπεδο του γράφου. Σε κάθε επίπεδο $l$, εφαρμόζεται \selectlanguage{english}greedy search,\selectlanguage{greek} κατά την οποία ο τρέχων κόμβος ενημερώνεται επιλέγοντας τον γείτονα που ελαχιστοποιεί την απόσταση από το ερώτημα $q$:
\begin{equation}
    c \leftarrow \arg\min_{v \in \mathcal{N}_l(c)} d(q,v)
\end{equation}
όπου $\mathcal{N}_l(c)$ είναι το σύνολο των γειτόνων του $c$ στο επίπεδο $l$. Η διαδικασία επαναλαμβάνεται μέχρι να μην υπάρχει γείτονας που να βελτιώνει την απόσταση από το $q$ --- δηλαδή μέχρι ο $c$ να αποτελεί τοπικό ελάχιστο στο τρέχον επίπεδο. Τότε ο $c$ χρησιμοποιείται ως κόμβος εκκίνησης στο αμέσως κατώτερο επίπεδο $l-1$, και η greedy αναζήτηση επαναλαμβάνεται εκεί από την αρχή.

Σκοπός αυτού του σταδίου είναι ο γρήγορος εντοπισμός μιας περιοχής του γράφου κοντά στο ερώτημα, χωρίς εκτενή διερεύνηση. Η ακρίβεια δεν αποτελεί προτεραιότητα εδώ --- το ζητούμενο είναι η αποδοτική σύγκλιση προς τη σωστή περιοχή.

\textbf{Παράδειγμα.} Έστω ο γράφος του Σχήματος~\ref{fig:hnsw_greedy_search} με ερώτημα $q$ και κόμβους $ep$, $A$, $B$ στο $L_2$, όπου $d(q,ep) > d(q,A) > d(q,B)$.
\begin{enumerate}
    \item \textbf{Βήμα 1:} Αρχικός κόμβος $c = ep$. Γείτονας $A$: $d(q,A) < d(q,ep)$ --- μετακίνηση, $c \leftarrow A$.
    \item \textbf{Βήμα 2:} $c = A$. Γείτονας $B$: $d(q,B) < d(q,A)$ --- μετακίνηση, $c \leftarrow B$.
    \item \textbf{Βήμα 3:} $c = B$. Ο μόνος γείτονας ($A$) έχει $d(q,A) > d(q,B)$ --- τοπικό ελάχιστο. Κατάβαση στο $L_1$ με αρχικό κόμβο $B$.

\subsubsection{Αναζήτηση στο επίπεδο βάσης (\selectlanguage{english}level 0\selectlanguage{greek})}
Στο επίπεδο βάσης η αναζήτηση διατηρεί δύο παράλληλες δομές: η ουρά υποψηφίων (\selectlanguage{english}candidate queue\selectlanguage{greek}) περιέχει κόμβους προς επέκταση, ενώ το σύνολο αποτελεσμάτων (\selectlanguage{english}top set\selectlanguage{greek}) αποθηκεύει τους έως $ef_{\text{search}}$ καλύτερους κόμβους που έχουν εξεταστεί. Μεγαλύτερο $ef_{\text{search}}$ αυξάνει την ακρίβεια με αντίτιμο το υπολογιστικό κόστος.

Σε κάθε βήμα, ο υποψήφιος με τη μικρότερη απόσταση από το $q$ αφαιρείται από την ουρά και εξερευνώνται οι γείτονές του: κάθε γείτονας που δεν έχει επισκεφθεί ακόμα εισάγεται στο \selectlanguage{english}top set\selectlanguage{greek} αν η απόστασή του από το $q$ είναι μικρότερη από τη μεγαλύτερη απόσταση που υπάρχει ήδη εκεί.

Η διαδικασία τερματίζεται όταν ο καλύτερος διαθέσιμος υποψήφιος προς επέκταση είναι χειρότερος από τον χειρότερο κόμβο στο \selectlanguage{english}top set\selectlanguage{greek}, δηλαδή όταν:
\[
d\bigl(q,v_{\text{best cand}}\bigr) > d\bigl(q,v_{\text{worst in top}}\bigr)
\]
Ο κανόνας αυτός εξασφαλίζει ότι δεν υπάρχουν πλέον υποψήφιοι ικανοί να βελτιώσουν το \selectlanguage{english}top set\selectlanguage{greek} --- κάθε εναπομείνας υποψήφιος απέχει από το $q$ περισσότερο από τον χειρότερο κόμβο που ήδη περιέχεται σε αυτό.

\subsubsection{Εξαγωγή αποτελεσμάτων}
Με την ολοκλήρωση της αναζήτησης στο επίπεδο βάσης, οι υποψήφιοι κόμβοι ταξινομούνται ως προς την απόστασή τους από το ερώτημα $q$ και επιστρέφονται οι $k$ πλησιέστεροι ως αποτέλεσμα.

Η παράμετρος $ef_{\text{search}}$ ρυθμίζει άμεσα τον συμβιβασμό ακρίβειας--ταχύτητας: μεγαλύτερες τιμές οδηγούν σε υψηλότερο \selectlanguage{english}recall\selectlanguage{greek} αλλά αυξάνουν αντίστοιχα τον αριθμό υπολογισμών απόστασης. Η σχέση αυτή είναι μονοτονική και αποτελεί τον κεντρικό άξονα του συμβιβασμού που εξετάζουμε στα επόμενα κεφάλαια.

\subsubsection{Ψευδοκώδικας Αλγορίθμων HNSW}

Οι Αλγόριθμοι~\ref{alg:hnsw_insert} και~\ref{alg:hnsw_search} παρουσιάζουν τυπικά τις δύο βασικές διαδικασίες του \selectlanguage{english}HNSW\selectlanguage{greek}. Η παρουσίαση ακολουθεί τη δομή της πρωτότυπης εργασίας \cite{malkov2018hnsw}, προσαρμοσμένη ώστε να αντιστοιχεί στην υλοποίησή μας.

\begin{algorithm}[H]
\selectlanguage{english}
\caption{HNSW Insert}
\label{alg:hnsw_insert}
\begin{algorithmic}[1]
\Require new element $q$, graph $G$, parameters $M$, $ef_{\text{construction}}$
\Ensure updated graph $G$ with $q$ inserted
\State $L \leftarrow$ level assigned to $q$ via geometric distribution: $L = \lfloor -\ln(\text{uniform}(0,1)) \cdot \frac{1}{\ln M} \rfloor$
\State $ep \leftarrow$ entry point of $G$ (node at top layer)
\For{$l = \text{top\_layer}(G)$ \textbf{down to} $L+1$}
    \State $ep \leftarrow$ \Call{GreedySearch}{$q$, $ep$, layer $l$, $ef=1$}
\EndFor
\For{$l = \min(L, \text{top\_layer}(G))$ \textbf{down to} $0$}
    \State $W \leftarrow$ \Call{BestFirstSearch}{$q$, $ep$, layer $l$, $ef_{\text{construction}}$}
    \State $\text{neighbors} \leftarrow$ \Call{SelectNeighbors}{$q$, $W$, $M$}
    \State connect $q$ to each node in $\text{neighbors}$ at layer $l$ (bidirectional)
    \For{each $e \in \text{neighbors}$}
        \If{$|\mathcal{N}_l(e)| > M$}
            \State $\mathcal{N}_l(e) \leftarrow$ \Call{SelectNeighbors}{$e$, $\mathcal{N}_l(e)$, $M$} \Comment{pruning}
        \EndIf
    \EndFor
    \State $ep \leftarrow W$
\EndFor
\If{$L > \text{top\_layer}(G)$}
    \State update entry point of $G$ to $q$
\EndIf
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\selectlanguage{english}
\caption{HNSW Search}
\label{alg:hnsw_search}
\begin{algorithmic}[1]
\Require query $q$, graph $G$, parameters $k$, $ef_{\text{search}}$
\Ensure set of $k$ approximate nearest neighbors of $q$
\State $ep \leftarrow$ entry point of $G$
\For{$l = \text{top\_layer}(G)$ \textbf{down to} $1$}
    \State $ep \leftarrow$ \Call{GreedySearch}{$q$, $ep$, layer $l$, $ef=1$}
\EndFor
\State $W \leftarrow$ \Call{BestFirstSearch}{$q$, $ep$, layer $0$, $ef_{\text{search}}$}
\State \Return top $k$ elements of $W$ sorted by $d(q, \cdot)$
\end{algorithmic}
\end{algorithm}

\selectlanguage{greek}
Στη συνάρτηση \selectlanguage{english}BestFirstSearch\selectlanguage{greek} διατηρούνται δύο δομές: μια ουρά υποψηφίων (\selectlanguage{english}min-heap\selectlanguage{greek} ως προς απόσταση) και ένα σύνολο αποτελεσμάτων μεγέθους $ef_{\text{search}}$ (\selectlanguage{english}max-heap\selectlanguage{greek}). Η εκτέλεση τερματίζει όταν ο καλύτερος υποψήφιος προς επέκταση είναι χειρότερος από τον χειρότερο κόμβο στο σύνολο αποτελεσμάτων. Η συνάρτηση \selectlanguage{english}SelectNeighbors\selectlanguage{greek} εφαρμόζει ευρετική επιλογής γειτόνων που λαμβάνει υπόψη και την ποικιλομορφία κατεύθυνσης: απορρίπτει έναν υποψήφιο $u$ αν υπάρχει ήδη επιλεγμένος γείτονας $w$ τέτοιος ώστε $d(u, w) < d(u, q)$ --- δηλαδή ο $w$ βρίσκεται πλησιέστερα στον $u$ από ό,τι το ερώτημα $q$, πράγμα που σημαίνει ότι ο $u$ και ο $w$ βρίσκονται στην ίδια «κατεύθυνση» ως προς το $q$ και ο $u$ είναι πλεονάζων.

Ο Αλγόριθμος~\ref{alg:select_neighbors} παρουσιάζει αυτή τη διαδικασία τυπικά. Η ευρετική ποικιλομορφίας είναι θεμελιώδης για την ποιότητα του γράφου: γράφοι κατασκευασμένοι χωρίς αυτήν τείνουν να έχουν πλεονάζουσες ακμές σε πυκνές περιοχές, υποβαθμίζοντας τη συνδεσιμότητα σε αραιότερες.

\begin{algorithm}[H]
\selectlanguage{english}
\caption{SelectNeighbors with Diversity Heuristic}
\label{alg:select_neighbors}
\begin{algorithmic}[1]
\Require element $q$, candidate set $W$, max neighbors $M$
\Ensure set $R$ of up to $M$ diverse neighbors
\State $R \leftarrow \emptyset$, $W_d \leftarrow W$ sorted by $d(q,\cdot)$ ascending
\For{each candidate $u \in W_d$}
    \If{$|R| \geq M$}
        \State \textbf{break}
    \EndIf
    \State $\text{dominated} \leftarrow \textbf{false}$
    \For{each $w \in R$}
        \If{$d(u, w) < d(u, q)$} \Comment{$u$ is closer to some selected $w$ than to $q$}
            \State $\text{dominated} \leftarrow \textbf{true}$; \textbf{break}
        \EndIf
    \EndFor
    \If{\textbf{not} dominated}
        \State $R \leftarrow R \cup \{u\}$
    \EndIf
\EndFor
\State \Return $R$
\end{algorithmic}
\end{algorithm}

\selectlanguage{greek}
Το κριτήριο στη γραμμή~8 ερμηνεύεται γεωμετρικά: ο $u$ δεν προστίθεται αν βρίσκεται «πίσω» από κάποιον ήδη επιλεγμένο κόμβο $w$ ως προς το $q$, δηλαδή αν ο $w$ «καλύπτει» αποτελεσματικά τη γειτονιά του $u$. Μια τέτοια ακμή $q \to u$ θα ήταν αλγοριθμικά πλεονάζουσα, καθώς κατά την αναζήτηση μπορεί να φτάσει κανείς στον $u$ μέσω του $w$ χωρίς άμεσο σύνδεσμο.

\section{Προσαρμοστικές Στρατηγικές Αναζήτησης στον HNSW}

\subsection{Το Πρόβλημα της Σταθερής Παραμετροποίησης}
Η βασική υλοποίηση HNSW αντιμετωπίζει κάθε ερώτημα με πανομοιότυπο τρόπο: η παράμετρος $ef_{\text{search}}$ είναι σταθερή, ανεξάρτητα από τη δυσκολία του ερωτήματος. Η παρατήρηση αυτή οδηγεί σε δύο αντιθετικά προβλήματα:
\begin{itemize}
    \item \textbf{Υπερβολική εξερεύνηση (over-searching):} για ερωτήματα που συγκλίνουν γρήγορα, η αναζήτηση συνεχίζεται άσκοπα --- εκτελούνται υπολογισμοί απόστασης που δεν μεταβάλλουν το τελικό αποτέλεσμα, αυξάνοντας το κόστος χωρίς κέρδος σε ακρίβεια.
    \item \textbf{Ανεπαρκής εξερεύνηση (under-searching):} για δύσκολα ερωτήματα, μια χαμηλή σταθερή τιμή $ef_{\text{search}}$ τερματίζει την αναζήτηση πριν εντοπιστούν οι πραγματικά κοντινότεροι γείτονες, με άμεσο αντίκτυπο στο recall.
\end{itemize}
Η ετερογένεια δυσκολίας μεταξύ ερωτημάτων καθιστά αδύνατη τη βέλτιστη ρύθμιση μιας ενιαίας τιμής: οποιαδήποτε σταθερή επιλογή $ef_{\text{search}}$ υπεραπλουστεύει την κατανομή των απαιτήσεων αναζήτησης.

Ο στόχος διατυπώνεται τυπικά ως πρόβλημα βελτιστοποίησης με περιορισμό: ελαχιστοποίηση του υπολογιστικού κόστους ανά ερώτημα υπό την προϋπόθεση ότι η ανάκληση \selectlanguage{english}(Recall)\selectlanguage{greek} δεν πέφτει κάτω από ένα αποδεκτό κατώφλι $\tau$:
\[
\min \; \text{Cost}(q) \quad \text{υπό τον περιορισμό}\quad \text{Recall@k}(q) \ge \tau
\]
όπου το κόστος μετριέται σε αριθμό υπολογισμών απόστασης. Η διατύπωση αυτή μετατοπίζει το κέντρο βάρους: το ζητούμενο δεν είναι η μεγιστοποίηση της ανάκλησης ανεξαρτήτως κόστους, αλλά η επίτευξη επαρκούς ανάκλησης με ελάχιστους υπολογισμούς — ερώτημα άμεσα σχετικό με τις απαιτήσεις παραγωγικών συστημάτων.

Η ίδια η εκτέλεση της αναζήτησης παράγει πληροφορία για το πόσο δύσκολο είναι ένα ερώτημα. Οι τρεις μέθοδοι που εξετάζονται στο κεφάλαιο αυτό αξιοποιούν διαφορετικά σήματα:
\begin{itemize}
    \item τη σταθεροποίηση του συνόλου των $k$ καλύτερων γειτόνων (\selectlanguage{english}top-$k$\selectlanguage{greek}) κατά τη διάρκεια της αναζήτησης --- αυτό αξιοποιεί η \selectlanguage{english}PiP\selectlanguage{greek}
    \item χαρακτηριστικά της κατάστασης αναζήτησης κατά τη διάρκεια εκτέλεσης, μέσω εκπαιδευμένου μοντέλου --- αυτό αξιοποιεί η \selectlanguage{english}DARTH\selectlanguage{greek}
    \item χαρακτηριστικά που εξάγονται από τα ανώτερα επίπεδα \emph{πριν} την εκτέλεση, για εκτίμηση της απαιτούμενης τιμής \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek} --- αυτό αξιοποιεί η \selectlanguage{english}Ada-ef\selectlanguage{greek}
\end{itemize}
Και στις τρεις περιπτώσεις, ο στόχος είναι να αποφύγουμε υπολογισμούς που δεν συνεισφέρουν στο τελικό αποτέλεσμα.

\subsection{Κατηγορίες Προσαρμοστικών Στρατηγικών}
Οι προσαρμοστικές στρατηγικές για τον HNSW μπορούν να ταξινομηθούν σε τρεις κατηγορίες, ανάλογα με το \emph{πότε} και \emph{πώς} αξιοποιούν τα σήματα δυσκολίας:

\begin{itemize}
    \item \textbf{Προσαρμοστικό $ef_{\text{search}}$ (Adaptive Budget Allocation):} Η παράμετρος $ef_{\text{search}}$ ρυθμίζεται per-query \emph{πριν} την εκτέλεση, με βάση εκτίμηση της δυσκολίας του ερωτήματος από χαρακτηριστικά που εξάγονται στα ανώτερα επίπεδα του γράφου. Ερωτήματα με υψηλή εκτιμώμενη δυσκολία λαμβάνουν μεγαλύτερο budget· «εύκολα» ερωτήματα εκτελούνται με χαμηλότερο $ef_{\text{search}}$.
    \item \textbf{Πρόωρος Τερματισμός (Early Termination):} Η αναζήτηση ξεκινά με σταθερό $ef_{\text{search}}$ και τερματίζεται \emph{κατά τη διάρκεια} της εκτέλεσης μόλις ανιχνευτεί σύγκλιση, αξιοποιώντας online χαρακτηριστικά της κατάστασης αναζήτησης.
    \item \textbf{Πρόωρος Τερματισμός μέσω Κορεσμού (\selectlanguage{english}Saturation-based Early Termination\selectlanguage{greek}):} Παραλλαγή του πρόωρου τερματισμού που δεν χρησιμοποιεί εκπαιδευμένο μοντέλο. Αντ' αυτού, παρακολουθεί τη σταθεροποίηση του \selectlanguage{english}top-$k$\selectlanguage{greek} συνόλου και τερματίζει όταν αυτό δεν μεταβάλλεται για $\Delta$ συνεχόμενες επαναλήψεις.
\end{itemize}

Οι τρεις μέθοδοι που αξιολογούνται κατανέμονται ως εξής: η \selectlanguage{english}DARTH\selectlanguage{greek} και η \selectlanguage{english}PiP\selectlanguage{greek} ανήκουν στον πρόωρο τερματισμό --- με διαφορετικό κριτήριο η καθεμία --- ενώ η \selectlanguage{english}Ada-ef\selectlanguage{greek} υλοποιεί προσαρμοστική ρύθμιση \selectlanguage{english}$ef_{\text{search}}$\selectlanguage{greek}. Η ποικιλία μηχανισμών παρέχει ολοκληρωμένη κάλυψη του χώρου λύσεων.

\subsection{DARTH: Adaptive Early Termination}

\subsubsection*{Στόχος}
Η DARTH (Declarative Recall Through early termination for HNSW) \cite{darth2025} στοχεύει στην ελαχιστοποίηση του αριθμού εξερευνώμενων κόμβων ανά ερώτημα, τερματίζοντας την αναζήτηση μόλις ένα εκπαιδευμένο μοντέλο εκτιμήσει ότι το τελικό recall θα υπερβεί τον στόχο $R_t$.

\subsubsection*{Κεντρική Ιδέα}
Η DARTH αντιμετωπίζει την προσαρμογή ως πρόβλημα ταξινόμησης: κατά τη διάρκεια της αναζήτησης, ένα μοντέλο \selectlanguage{english}LightGBM\selectlanguage{greek} εκπαιδευμένο \selectlanguage{english}offline\selectlanguage{greek} αξιολογεί χαρακτηριστικά της τρέχουσας κατάστασης και αποφασίζει αν μπορεί να τερματιστεί νωρίτερα. Το \selectlanguage{english}LightGBM\selectlanguage{greek} είναι ένας ταχύς αλγόριθμος ενισχυτικής μάθησης με κλιμακούμενα δέντρα αποφάσεων (\selectlanguage{english}gradient boosted decision trees\selectlanguage{greek}), κατάλληλος για ταξινόμηση πινακοειδών χαρακτηριστικών. Επιλέχθηκε διότι: (α) η εκτίμηση (inference) εκτελείται σε μικροδευτερόλεπτα, χωρίς να επιβαρύνει σημαντικά το κόστος αναζήτησης, (β) τα χαρακτηριστικά κατάστασης που χρησιμοποιεί η \selectlanguage{english}DARTH\selectlanguage{greek} (αποστάσεις, πλήθος υποψηφίων κ.λπ.) είναι δομημένα αριθμητικά δεδομένα --- ακριβώς ο τύπος εισόδου για τον οποίο τα μοντέλα \selectlanguage{english}gradient boosting\selectlanguage{greek} υπερτερούν έναντι νευρωνικών δικτύων.

Τα χαρακτηριστικά που εξάγονται σε κάθε σημείο ελέγχου $t$ (κάθε $\delta$ βήματα) περιλαμβάνουν:
\begin{itemize}
    \item $\Delta d_{\min}(t) = d_{\min}(t-\delta) - d_{\min}(t)$: βελτίωση της καλύτερης απόστασης από το προηγούμενο σημείο ελέγχου
    \item $s_{\text{top-}k}(t)$: σταθερότητα top-$k$ συνόλου — ποσοστό κόμβων που παρέμειναν στους $k$ καλύτερους μεταξύ δύο ελέγχων
    \item $\Delta t_{\text{last}}$: αριθμός βημάτων από την τελευταία βελτίωση του top-$k$
    \item $r_W(t) = |W|/ef_{\text{search}}$: αναλογία πλήρωσης του συνόλου αποτελεσμάτων $W$ --- το $W$ κρατά τους $ef_{\text{search}}$ καλύτερους κόμβους που έχουν εξεταστεί ως τώρα· όταν $r_W = 1$ το σύνολο είναι γεμάτο και ο τερματισμός γίνεται ενεργός
    \item $d_k(t)/d_1(t)$: λόγος $k$-οστής προς καλύτερης απόστασης — μικρός λόγος υποδηλώνει πυκνή γειτονιά
\end{itemize}
Για κάθε ερώτημα παράγονται $\lfloor ef_{\text{search}} / \delta \rfloor$ διανύσματα $\mathbf{x}(t)$ --- ένα ανά σημείο ελέγχου. Κατά την \textbf{εκπαίδευση}, κάθε τέτοιο διάνυσμα αποτελεί ένα ξεχωριστό δείγμα με ετικέτα $1$ αν το τελικό recall της συγκεκριμένης αναζήτησης ήταν $\geq R_t$, αλλιώς $0$. Κατά την \textbf{αναζήτηση}, το μοντέλο καλείται σε κάθε σημείο ελέγχου και επιστρέφει εκτίμηση $\hat{R}(t)$· μόλις $\hat{R}(t) \geq R_t = 0.90$, η αναζήτηση τερματίζεται αμέσως και δεν παράγονται περαιτέρω διανύσματα. Το μοντέλο εκπαιδεύεται ξεχωριστά για κάθε συνδυασμό $(dataset, M, ef_{\text{construction}}, k)$, διότι τα σήματα δυσκολίας εξαρτώνται από τη δομή του γράφου και την κατανομή των δεδομένων.

\begin{algorithm}[H]
\selectlanguage{english}
\caption{DARTH: Search with Learned Early Termination}
\label{alg:darth}
\begin{algorithmic}[1]
\Require query $q$, graph $G$, predictor $f_\theta$, parameters $k$, $ef_{\text{search}}$, $R_t$
\Ensure approximate $k$-NN of $q$
\State $ep \leftarrow$ greedy navigation to layer 0 (as in standard HNSW)
\State $W \leftarrow \{ep\}$, $C \leftarrow \{ep\}$ \Comment{results set, candidate queue}
\State $\text{step} \leftarrow 0$
\While{$C \neq \emptyset$}
    \State $c \leftarrow \arg\min_{v \in C} d(q, v)$; $C \leftarrow C \setminus \{c\}$
    \State expand $c$: add unvisited neighbors to $C$ and $W$ if they improve $W$
    \State $\text{step} \mathrel{+}= 1$
    \If{$\text{step} \bmod \delta = 0$} \Comment{check every $\delta$ steps}
        \State $\mathbf{x} \leftarrow$ \Call{ExtractFeatures}{$W$, $\text{step}$}
        \If{$f_\theta(\mathbf{x}) \geq R_t$}
            \State \textbf{break} \Comment{early termination}
        \EndIf
    \EndIf
\EndWhile
\State \Return top $k$ of $W$
\end{algorithmic}
\end{algorithm}

\selectlanguage{greek}
Ο αλγόριθμος εκκινεί (γραμμή~1) με \selectlanguage{english}greedy\selectlanguage{greek} πλοήγηση στα ανώτερα επίπεδα, όπως στο τυπικό \selectlanguage{english}HNSW\selectlanguage{greek}, για να εντοπίσει έναν κόμβο εισόδου $ep$ στο επίπεδο βάσης. Στη συνέχεια (γραμμή~2) αρχικοποιούνται δύο δομές: το σύνολο αποτελεσμάτων $W$, το οποίο κρατά τους $ef_{\text{search}}$ πλησιέστερους κόμβους που έχουν βρεθεί, και η ουρά υποψηφίων $C$. Ο κύριος βρόχος (γραμμές~4--13) εκτελεί \selectlanguage{english}best-first search\selectlanguage{greek}: σε κάθε επανάληψη αφαιρείται από το $C$ ο κόμβος $c$ με τη μικρότερη απόσταση από το $q$ (γραμμή~5) και εξερευνώνται οι γείτονές του, οι οποίοι εισάγονται στο $W$ αν βελτιώνουν τα τρέχοντα αποτελέσματα (γραμμή~6). Κάθε $\delta$ βήματα (γραμμή~8) εξάγεται το διάνυσμα χαρακτηριστικών $\mathbf{x}$ από την τρέχουσα κατάσταση του $W$ (γραμμή~9) και τροφοδοτείται στο μοντέλο $f_\theta$. Αν η εκτιμώμενη τιμή \selectlanguage{english}recall\selectlanguage{greek} ξεπεράσει το κατώφλι $R_t$ (γραμμή~10), η αναζήτηση τερματίζεται πρόωρα (γραμμή~11) χωρίς να εξαντληθεί ο προϋπολογισμός $ef_{\text{search}}$. Διαφορετικά, συνεχίζεται κανονικά και στο τέλος επιστρέφονται οι $k$ πλησιέστεροι κόμβοι του $W$ (γραμμή~15).

\textbf{Παράδειγμα λειτουργίας \selectlanguage{english}DARTH\selectlanguage{greek}.}
Έστω ερώτημα $q = (6,3)$ στον γράφο του §\ref{sec:hnsw_example} με $ef_{\text{search}} = 10$, $k = 1$ και $\delta = 5$ (έλεγχος ανά 5 βήματα). Στο βήμα $t = 5$ έχουν εξεταστεί 5 κόμβοι και το σύνολο αποτελεσμάτων $W$ έχει σταθεροποιηθεί γύρω από τον $F = (6,2)$. Εξάγεται το διάνυσμα χαρακτηριστικών:

\begin{itemize}
    \item $\Delta d_{\min} = d_{\min}(t{=}0) - d_{\min}(t{=}5) = 3.16 - 1.00 = 2.16$ \quad (μεγάλη βελτίωση: εύκολο ερώτημα)
    \item $s_{\text{top-}k} = 1.00$ \quad (το top-1 δεν άλλαξε από το βήμα 2)
    \item $\Delta t_{\text{last}} = 3$ \quad (τελευταία βελτίωση πριν 3 βήματα)
    \item $r_W = 5/10 = 0.50$ \quad (μισές θέσεις $ef$ γεμάτες)
    \item $d_k / d_1 = 1.00 / 1.00 = 1.00$ \quad (μόνο 1 αποτέλεσμα, λόγος = 1)
\end{itemize}

Το LightGBM λαμβάνει το διάνυσμα $\mathbf{x} = (2.16,\; 1.00,\; 3,\; 0.50,\; 1.00)$ και επιστρέφει $\hat{R} = 0.94 \geq R_t = 0.90$ → πρόωρος τερματισμός. Αντίθετα, σε ένα δύσκολο ερώτημα όπου $\Delta d_{\min} = 0.02$ και $s_{\text{top-}k} = 0.40$, η εκτίμηση θα ήταν $\hat{R} = 0.71 < 0.90$ και η αναζήτηση θα συνεχιζόταν.

\subsubsection*{Πλεονεκτήματα}
\begin{itemize}
    \item Άμεσος ορισμός στόχου \selectlanguage{english}recall\selectlanguage{greek}: ο χρήστης ορίζει το επιθυμητό $R_t$ και ο αλγόριθμος τερματίζει αυτόματα όταν το εκτιμώμενο \selectlanguage{english}recall\selectlanguage{greek} τον ικανοποιεί, χωρίς να χρειάζεται χειροκίνητη ρύθμιση παραμέτρων τερματισμού.
    \item Δεν τροποποιεί τον γράφο ούτε τον πυρήνα της αναζήτησης· ενσωματώνεται σε υπάρχουσα υλοποίηση \selectlanguage{english}HNSW\selectlanguage{greek} προσθέτοντας μόνο τον έλεγχο τερματισμού εντός του βρόχου αναζήτησης.
    \item Αποδοτικό σε datasets με ετερογενή κατανομή δυσκολίας, όπου σημαντικό ποσοστό ερωτημάτων συγκλίνει γρήγορα.
\end{itemize}

\subsubsection*{Περιορισμοί}
\begin{itemize}
    \item Απαιτεί \selectlanguage{english}offline\selectlanguage{greek} εκπαίδευση ξεχωριστού μοντέλου \selectlanguage{english}LightGBM\selectlanguage{greek} για κάθε συνδυασμό $(dataset,\; M,\; ef_C,\; k)$, διότι τα στατιστικά χαρακτηριστικά της αναζήτησης (κατανομή αποστάσεων, ρυθμός σύγκλισης \selectlanguage{english}top-$k$\selectlanguage{greek} κ.λπ.) εξαρτώνται από τη δομή του εκάστοτε γράφου και την κατανομή του dataset· ένα μοντέλο εκπαιδευμένο σε έναν συνδυασμό δεν γενικεύεται σε άλλον.
    \item Η ακρίβεια του μοντέλου πρόβλεψης εξαρτάται από το πόσο αντιπροσωπευτικό είναι το σύνολο εκπαίδευσης: αν τα ερωτήματα παραγωγής ακολουθούν διαφορετική κατανομή από αυτά με τα οποία εκπαιδεύτηκε το μοντέλο, η εκτίμηση $\hat{R}$ μπορεί να αποκλίνει σημαντικά από την πραγματική τιμή \selectlanguage{english}recall\selectlanguage{greek}.
    \item Σε datasets με ομοιόμορφη κατανομή δυσκολίας (π.χ.\ GloVe-100), το predictor έχει περιορισμένη πληροφορία διαχωρισμού και το κέρδος γίνεται αμελητέο.
\end{itemize}

\subsubsection*{Υπολογιστικό Κόστος}
Σε κάθε σημείο ελέγχου (κάθε $\delta$ βήματα), η εξαγωγή χαρακτηριστικών έχει κόστος $O(k)$ (ανάγνωση του συνόλου $W$). Η συμπερασματική φάση του LightGBM εκτελείται σε σταθερό χρόνο ανάλογο με τον αριθμό δέντρων και το βάθος τους. Συνολικά, το overhead ανά ερώτημα παραμένει κάτω από 5\% σε σχέση με τον baseline, καθώς ο αριθμός σημείων ελέγχου είναι $O(ef_{\text{search}}/\delta)$.

\subsection{PiP: Πρόωρος Τερματισμός βάσει Κορεσμού}

\subsubsection*{Στόχος}
Η \selectlanguage{english}PiP (Patience in Proximity)\selectlanguage{greek} \cite{pip2025} στοχεύει στην ελαχιστοποίηση του αριθμού εξερευνώμενων κόμβων ανά ερώτημα, τερματίζοντας την αναζήτηση μόλις το σύνολο των $k$ καλύτερων γειτόνων σταθεροποιηθεί --- δηλαδή παύει να μεταβάλλεται για επαρκώς πολλές συνεχόμενες επαναλήψεις.

\subsubsection*{Κεντρική Ιδέα}
Η \selectlanguage{english}PiP\selectlanguage{greek} βασίζεται στην παρατήρηση ότι κατά τη διάρκεια της \selectlanguage{english}HNSW\selectlanguage{greek} αναζήτησης, το σύνολο των $k$ καλύτερων γειτόνων τείνει να σταθεροποιείται πολύ πριν εξαντληθεί το \selectlanguage{english}budget\selectlanguage{greek} $ef_{\text{search}}$. Αντί να τερματίζει βάσει στατικής παραμέτρου ή \selectlanguage{english}ML predictor\selectlanguage{greek}, η \selectlanguage{english}PiP\selectlanguage{greek} παρακολουθεί διαδοχικά «στιγμιότυπα» του \selectlanguage{english}top-$k$\selectlanguage{greek} συνόλου και τερματίζει όταν αυτό δεν αλλάζει για $\Delta$ συνεχόμενες επαναλήψεις («\selectlanguage{english}patience\selectlanguage{greek}»).

Η κεντρική ποσότητα που παρακολουθείται είναι:
\begin{equation}
    \phi_{h,l}(q) = 100 \cdot \frac{|N_{h-1,l}(q) \cap N_{h,l}(q)|}{k}
\end{equation}
όπου $N_{h,l}(q)$ είναι το σύνολο των $k$ κοντινότερων γειτόνων που έχουν συλλεχθεί μετά από $h$ επεκτάσεις στο επίπεδο $l$. Όταν $\phi_{h,l}(q) \geq \gamma$ (π.χ.\ $\gamma = 95$) για $\Delta$ συνεχόμενες επαναλήψεις, η αναζήτηση τερματίζεται.

\begin{algorithm}[H]
\selectlanguage{english}
\caption{PiP: Patience in Proximity Search}
\label{alg:pip}
\begin{algorithmic}[1]
\Require query $q$, graph $G$, parameters $k$, $ef_{\text{search}}$, $\gamma$, $\Delta$
\Ensure approximate $k$-NN of $q$
\State $ep \leftarrow$ greedy navigation to layer 0
\State $W \leftarrow \{ep\}$, $C \leftarrow \{ep\}$
\State $N_{\text{prev}} \leftarrow \emptyset$, $\text{sat\_counter} \leftarrow 0$
\While{$C \neq \emptyset$}
    \State $c \leftarrow \arg\min_{v \in C} d(q, v)$; $C \leftarrow C \setminus \{c\}$
    \State expand $c$: add unvisited neighbors to $C$ and $W$ (standard HNSW step)
    \State $N_{\text{curr}} \leftarrow $ top-$k$ of $W$
    \If{$|N_{\text{curr}}| = k$ \textbf{and} $N_{\text{prev}} \neq \emptyset$}
        \State $\phi_h \leftarrow 100 \cdot |N_{\text{prev}} \cap N_{\text{curr}}| \;/\; k$
        \If{$\phi_h \geq \gamma$}
            \State $\text{sat\_counter} \mathrel{+}= 1$
        \Else
            \State $\text{sat\_counter} \leftarrow 0$
        \EndIf
        \If{$\text{sat\_counter} \geq \Delta$}
            \State \textbf{break} \Comment{patience threshold reached}
        \EndIf
    \EndIf
    \State $N_{\text{prev}} \leftarrow N_{\text{curr}}$
\EndWhile
\State \Return top $k$ of $W$
\end{algorithmic}
\end{algorithm}

\selectlanguage{greek}
Ο αλγόριθμος εκκινεί (γραμμή~1) με \selectlanguage{english}greedy\selectlanguage{greek} πλοήγηση για εντοπισμό κόμβου εισόδου $ep$ στο επίπεδο βάσης, ακριβώς όπως στην τυπική αναζήτηση \selectlanguage{english}HNSW\selectlanguage{greek}. Στη γραμμή~2 αρχικοποιούνται το σύνολο αποτελεσμάτων $W$ και η ουρά υποψηφίων $C$, ενώ στη γραμμή~3 μηδενίζεται ο μετρητής κορεσμού $\text{sat\_counter}$ και το προηγούμενο \selectlanguage{english}top-$k$\selectlanguage{greek} ορίζεται κενό. Σε κάθε επανάληψη του κύριου βρόχου (γραμμές~4--16) αφαιρείται ο πλησιέστερος υποψήφιος $c$ (γραμμή~5) και εξερευνώνται οι γείτονές του (γραμμή~6). Ακολούθως (γραμμή~7) εξάγεται το τρέχον \selectlanguage{english}top-$k$\selectlanguage{greek} σύνολο $N_{\text{curr}}$. Εφόσον το $W$ έχει συγκεντρώσει τουλάχιστον $k$ κόμβους και υπάρχει προηγούμενο στιγμιότυπο (γραμμή~8), υπολογίζεται η αλληλοκάλυψη $\phi_h$ (γραμμή~9): αν $\phi_h \geq \gamma$, ο μετρητής αυξάνεται· αν όχι, μηδενίζεται (γραμμές~10--13). Όταν ο μετρητής φτάσει $\Delta$ (γραμμή~14), το \selectlanguage{english}top-$k$\selectlanguage{greek} θεωρείται κορεσμένο και η αναζήτηση τερματίζεται πρόωρα. Διαφορετικά συνεχίζεται κανονικά και επιστρέφονται οι $k$ καλύτεροι κόμβοι (γραμμή~18).

$\gamma$ & $\in [0, 100]$ & Κατώφλι κορεσμού: ελάχιστο ποσοστό αλληλοκάλυψης consecutive \selectlanguage{english}top-$k$\selectlanguage{greek} για να μετρηθεί επανάληψη ως «σταθερή» \\

\bottomrule
\end{tabular}
\end{center}

\textbf{Παράδειγμα λειτουργίας \selectlanguage{english}PiP\selectlanguage{greek}.}
Έστω ερώτημα $q = (6,3)$ στον γράφο του §\ref{sec:hnsw_example} με $ef_{\text{search}} = 10$, $k = 1$, $\gamma = 95$ και $\Delta = 3$.

Στα πρώτα βήματα εξερεύνησης, το \selectlanguage{english}top-1\selectlanguage{greek} αλλάζει: ξεκινά από τον $C$ (βήμα 1), βελτιώνεται στον $F = (6,2)$ (βήμα 3). Μόλις ο $F$ εισέλθει στο \selectlanguage{english}top-1\selectlanguage{greek} και παραμείνει αμετάβλητος:

\begin{itemize}
    \item Βήμα 4: $N_{\text{curr}} = \{F\}$, $N_{\text{prev}} = \{F\}$ $\Rightarrow$ $\phi_4 = 100\% \geq 95$ $\Rightarrow$ $\text{sat\_counter} = 1$
    \item Βήμα 5: $\phi_5 = 100\%$ $\Rightarrow$ $\text{sat\_counter} = 2$
    \item Βήμα 6: $\phi_6 = 100\%$ $\Rightarrow$ $\text{sat\_counter} = 3 = \Delta$ $\Rightarrow$ \textbf{τερματισμός}
\end{itemize}

Αποτέλεσμα: $F = (6,2)$, ο πραγματικός κοντινότερος γείτονας, εντοπίστηκε με 6 βήματα αντί 10. Αντίθετα, σε ένα δύσκολο ερώτημα όπου το \selectlanguage{english}top-$k$\selectlanguage{greek} συνεχίζει να αλλάζει, ο $\text{sat\_counter}$ δεν φτάνει το $\Delta$ και η αναζήτηση συνεχίζεται κανονικά έως εξαντλήσεως του $ef_{\text{search}}$.

\subsubsection*{Πλεονεκτήματα}
\begin{itemize}
    \item Δεν απαιτεί εκπαίδευση μοντέλου --- το κριτήριο κορεσμού είναι αναλυτικό και ανεξάρτητο του \selectlanguage{english}dataset\selectlanguage{greek}.
    \item Εφαρμόζεται ως επίθεμα στον βρόχο αναζήτησης χωρίς τροποποίηση του γράφου ή επιπλέον δομών δεδομένων.
    \item Αποδοτικό σε ερωτήματα που συγκλίνουν γρήγορα, που αποτελούν την πλειοψηφία σε \selectlanguage{english}datasets\selectlanguage{greek} με ετερογενή κατανομή δυσκολίας.
\end{itemize}

\subsubsection*{Περιορισμοί}
\begin{itemize}
    \item Σε \selectlanguage{english}datasets\selectlanguage{greek} με ομοιόμορφη κατανομή (π.χ.\ \selectlanguage{english}GloVe-100, SIFT-1M\selectlanguage{greek}), το \selectlanguage{english}top-$k$\selectlanguage{greek} σύνολο αλλάζει σταδιακά σε όλη τη διάρκεια αναζήτησης, οπότε το κατώφλι κορεσμού δεν ενεργοποιείται εγκαίρως --- οδηγώντας σε κορεσμό ανάκλησης.
    \item Αν ο $\gamma$ οριστεί πολύ υψηλά ή ο $\Delta$ πολύ μικρός, η αναζήτηση μπορεί να τερματίσει πρόωρα σε δύσκολα ερωτήματα με αποτέλεσμα πτώση \selectlanguage{english}recall\selectlanguage{greek}.
    \item Η παρακολούθηση του \selectlanguage{english}top-$k$\selectlanguage{greek} συνόλου σε κάθε βήμα εισάγει μικρό \selectlanguage{english}overhead\selectlanguage{greek} $O(k)$ ανά επέκταση.
\end{itemize}

\subsubsection*{Υπολογιστικό Κόστος}
Σε κάθε βήμα εξερεύνησης, η \selectlanguage{english}PiP\selectlanguage{greek} εκτελεί $O(k)$ πράξεις για τον υπολογισμό του $\phi_h$ (σύγκριση δύο συνόλων μεγέθους $k$). Το \selectlanguage{english}overhead\selectlanguage{greek} αυτό είναι αμελητέο σε σχέση με τον υπολογισμό απόστασης $d(q, \cdot)$, ιδίως για υψηλές διαστάσεις όπου κάθε τέτοιος υπολογισμός κοστίζει $O(d)$ πράξεις κινητής υποδιαστολής.

\subsection{Ada-ef: Adaptive efSearch Selection}

\subsubsection*{Στόχος}
Η Ada-ef (Distribution-Aware Adaptive ef) \cite{adaef2025} στοχεύει στη βέλτιστη κατανομή υπολογιστικού budget ανά ερώτημα: αντί να εκτελεί η ίδια αναζήτηση με σταθερό $ef_{\text{search}}$, επιλέγει εκ των προτέρων — για κάθε ερώτημα ξεχωριστά — τη μικρότερη τιμή $ef^*$ που αναμένεται να επιτύχει τον στόχο recall $R_t$.

\subsubsection*{Κεντρική Ιδέα}
Η Ada-ef ακολουθεί διαφορετική φιλοσοφία: αντί να τροποποιεί τη λογική αναζήτησης κατά τη διάρκειά της, αποφασίζει το κατάλληλο $ef_{\text{search}}$ πριν ξεκινήσει η αναζήτηση στο βασικό επίπεδο.

Η \selectlanguage{english}offline\selectlanguage{greek} ανάλυση απόδοσης, κατά την οποία κατασκευάζεται ο πίνακας αναφοράς, λειτουργεί ως εξής:
\begin{enumerate}
    \item Για κάθε τιμή $ef$ στο σύνολο $\mathcal{E} = \{ef_1, ef_2, \ldots, ef_L\}$, εκτελείται αναζήτηση σε σύνολο εκτίμησης $\mathcal{Q}_{\text{cal}}$ (δοκιμαστικά ερωτήματα με γνωστά αποτελέσματα) και υπολογίζεται το μέσο \selectlanguage{english}recall\selectlanguage{greek} $\bar{R}(ef)$.
    \item Για κάθε ερώτημα $q_i \in \mathcal{Q}_{\text{cal}}$ εξάγεται τιμή ομοιότητας $s_i$ — η μέση cosine similarity μεταξύ $q_i$ και των πρώτων κόμβων που συναντώνται κατά την greedy πλοήγηση στα ανώτερα επίπεδα.
    \item Κατασκευάζεται πίνακας lookup: $\text{LUT}[ef][s] \mapsto \hat{R}(ef, s)$, ο οποίος προβλέπει το recall για δεδομένο $ef$ και τιμή ομοιότητας $s$.
\end{enumerate}
Κατά την αναζήτηση, για κάθε νέο ερώτημα $q$:
\begin{enumerate}
    \item Εκτελείται \selectlanguage{english}greedy\selectlanguage{greek} πλοήγηση στα ανώτερα επίπεδα και υπολογίζεται $\hat{s}(q)$ --- η μέση \selectlanguage{english}cosine similarity\selectlanguage{greek} μεταξύ του $q$ και των κόμβων που συναντώνται κατά αυτή την πλοήγηση (ίδιος υπολογισμός με το βήμα~2 της \selectlanguage{english}offline\selectlanguage{greek} ανάλυσης).
    \item Επιλέγεται $ef^* = \min\{ef \in \mathcal{E} : \text{LUT}[ef][\hat{s}(q)] \geq R_t\}$.
    \item Εκτελείται αναζήτηση στο βασικό επίπεδο με $ef_{\text{search}} = ef^*$.
\end{enumerate}
Η προσέγγιση αποδίδει όταν η ομοιότητα $\hat{s}(q)$ προβλέπει αξιόπιστα τη δυσκολία του ερωτήματος — κάτι που ισχύει σε datasets με ευδιάκριτη ετερογένεια (text embeddings) αλλά όχι αναγκαία σε ομοιόμορφες κατανομές (GloVe-100). Επιπλέον, η χρήση cosine similarity ως μετρικής ομοιότητας (αντί της ευκλείδειας απόστασης που χρησιμοποιούν οι υπόλοιπες μέθοδοι) εισάγει ασυμμετρία που πρέπει να λαμβάνεται υπόψη κατά τη σύγκριση αποτελεσμάτων.

\textbf{Παράδειγμα λειτουργίας Ada-ef.}
Έστω $\mathcal{E} = \{50, 100, 200\}$, $R_t = 0.90$ και ο ακόλουθος lookup table (απλοποιημένος):

\begin{center}
\begin{tabular}{lccc}
\toprule
$\hat{s}$ bucket & $ef=50$ & $ef=100$ & $ef=200$ \\
\midrule
$[0.60, 0.70)$ & 0.78 & 0.86 & 0.94 \\
$[0.70, 0.80)$ & 0.82 & 0.91 & 0.97 \\
$[0.80, 0.90)$ & 0.88 & 0.95 & 0.99 \\
\bottomrule
\end{tabular}
\end{center}

Ένα νέο ερώτημα $q$ εκτελεί greedy πλοήγηση στα ανώτερα επίπεδα και παράγει εκτίμηση $\hat{s} = 0.82$ (bucket $[0.80, 0.90)$). Η Ada-ef σαρώνει τον πίνακα: $ef=50 \to 0.88 < 0.90$, $ef=100 \to 0.95 \geq 0.90$ — επιλέγεται $ef^* = 100$. Για ερώτημα με $\hat{s} = 0.62$ (bucket $[0.60, 0.70)$), ούτε $ef=50$ ούτε $ef=100$ επαρκούν· επιλέγεται $ef^* = 200$ (fallback). Η προσαρμογή εξοικονομεί υπολογιστικό κόστος στα εύκολα ερωτήματα χωρίς να θυσιάζει ακρίβεια.

\subsubsection*{Πλεονεκτήματα}
\begin{itemize}
    \item Ελάχιστο runtime overhead: η επιλογή $ef^*$ απαιτεί greedy πλοήγηση στα ανώτερα επίπεδα (λίγα βήματα) και αναζήτηση σε πίνακα σταθερού μεγέθους.
    \item Δεν τροποποιεί τον κύριο αλγόριθμο αναζήτησης — η αλλαγή αφορά αποκλειστικά την παράμετρο $ef_{\text{search}}$.
    \item Αποδοτικό όταν υπάρχει ισχυρή συσχέτιση μεταξύ $\hat{s}$ και πραγματικής δυσκολίας ερωτήματος.
\end{itemize}

\subsubsection*{Περιορισμοί}
\begin{itemize}
    \item Απαιτεί \selectlanguage{english}offline\selectlanguage{greek} ανάλυση σε αντιπροσωπευτικό σύνολο ερωτημάτων· αν τα ερωτήματα κατά την πραγματική χρήση ακολουθούν διαφορετική κατανομή από αυτή του συνόλου εκτίμησης, ο πίνακας αναφοράς χρειάζεται επανακατασκευή.
    \item Η χρήση \selectlanguage{english}cosine similarity\selectlanguage{greek} ως σήμα δυσκολίας εισάγει εννοιολογική ασυμμετρία: η μέθοδος σχεδιάστηκε για \selectlanguage{english}text embeddings\selectlanguage{greek}, όπου υψηλή ομοιότητα με τους πρώτους κόμβους που συναντά η greedy πλοήγηση σημαίνει ότι το ερώτημα βρίσκεται σε πυκνή περιοχή και άρα είναι εύκολο. Σε \selectlanguage{english}datasets\selectlanguage{greek} που χρησιμοποιούν ευκλείδεια απόσταση (π.χ.\ \selectlanguage{english}SIFT\selectlanguage{greek}), η \selectlanguage{english}cosine similarity\selectlanguage{greek} δεν έχει αντίστοιχη ερμηνεία: δύο διανύσματα μπορεί να έχουν υψηλή \selectlanguage{english}cosine similarity\selectlanguage{greek} (ίδια κατεύθυνση) αλλά μεγάλη ευκλείδεια απόσταση (διαφορετικό μέγεθος), οπότε το σήμα $\hat{s}(q)$ παύει να προβλέπει αξιόπιστα τη δυσκολία. Για εφαρμογή σε ευκλείδεια \selectlanguage{english}datasets\selectlanguage{greek}, θα χρειαζόταν επαναορισμός του χαρακτηριστικού $\hat{s}(q)$ βάσει ευκλείδειας ομοιότητας και πλήρης επανακατασκευή του πίνακα αναφοράς.
    \item Σε datasets χωρίς ετερογένεια δυσκολίας, η εκτίμηση $\hat{s}$ δεν διαχωρίζει εύκολα και δύσκολα ερωτήματα, και η Ada-ef συμπεριφέρεται ουσιαστικά ως baseline με σταθερό $ef$.
\end{itemize}

\subsubsection*{Υπολογιστικό Κόστος}
Offline: κατασκευή του LUT απαιτεί $|\mathcal{E}| \times |\mathcal{Q}_{\text{cal}}|$ πλήρεις αναζητήσεις. Online: η επίπλεον επεξεργασία ανά ερώτημα περιορίζεται σε greedy πλοήγηση $O(\log N)$ βημάτων και lookup σε πίνακα $O(|\mathcal{E}|)$ — πρακτικά αμελητέο σε σχέση με την αναζήτηση στο βασικό επίπεδο.

\subsection{Αναμενόμενη Συμπεριφορά στα Πειράματα}
Πριν από την παρουσίαση των αποτελεσμάτων, διατυπώνονται ρητές υποθέσεις σχετικά με την αναμενόμενη συμπεριφορά κάθε μεθόδου. Το Κεφάλαιο~5 αξιολογεί κατά πόσο αυτές οι υποθέσεις επιβεβαιώνονται εμπειρικά.

\textbf{Υπόθεση 1 — DARTH.}
Αναμένεται να αποδίδει καλύτερα σε datasets με ετερογενή κατανομή δυσκολίας ερωτημάτων (MS MARCO Text, SIFT Small), όπου σημαντικό ποσοστό ερωτημάτων συγκλίνει γρήγορα. Σε ομοιόμορφες κατανομές (\selectlanguage{english}GloVe-100\selectlanguage{greek}), το μοντέλο πρόβλεψης έχει περιορισμένη ικανότητα διαχωρισμού --- τα ερωτήματα έχουν παρόμοια δυσκολία, οπότε δεν υπάρχει αξιόπιστο σήμα για πρόωρο τερματισμό --- και το κέρδος θα είναι μικρό. Ο αριθμός υπολογισμών απόστασης αναμένεται να μειωθεί χωρίς αξιοσημείωτη απώλεια recall.

\textbf{Υπόθεση 2 — PiP.}
Αναμένεται να μειώνει τον αριθμό εξερευνώμενων κόμβων σε \selectlanguage{english}datasets\selectlanguage{greek} με ετερογενή κατανομή δυσκολίας, όπου τα εύκολα ερωτήματα έχουν \selectlanguage{english}top-$k$\selectlanguage{greek} που σταθεροποιείται γρήγορα. Σε \selectlanguage{english}datasets\selectlanguage{greek} με ομοιόμορφη κατανομή (\selectlanguage{english}SIFT-1M, GloVe-100\selectlanguage{greek}), το \selectlanguage{english}top-$k$\selectlanguage{greek} συνεχίζει να μεταβάλλεται σχεδόν μέχρι τέλους, οπότε το κριτήριο κορεσμού δεν ενεργοποιείται εγκαίρως --- οδηγώντας σε \selectlanguage{english}recall saturation\selectlanguage{greek}: αδυναμία ταυτόχρονης μείωσης κόστους και διατήρησης \selectlanguage{english}recall\selectlanguage{greek}.

\textbf{Υπόθεση 3 — Ada-ef.}
Αναμένεται να βελτιώνει την κατανομή υπολογιστικού budget όταν το $\hat{s}$ συσχετίζεται ισχυρά με τη δυσκολία ερωτήματος (text embeddings). Λόγω χρήσης \selectlanguage{english}cosine similarity\selectlanguage{greek}, η μέθοδος μπορεί να υστερεί σε \selectlanguage{english}datasets\selectlanguage{greek} που αξιολογούνται με ευκλείδεια απόσταση, καθώς το χαρακτηριστικό $\hat{s}(q)$ --- που ορίζεται ως \selectlanguage{english}cosine similarity\selectlanguage{greek} με τους πρώτους κόμβους της greedy πλοήγησης --- δεν προβλέπει αξιόπιστα τη δυσκολία του ερωτήματος υπό ευκλείδεια μετρική. Σε ομοιόμορφα datasets, η Ada-ef αναμένεται να συμπεριφέρεται ουσιαστικά ως baseline.

\textbf{Υπόθεση 4 — Εξάρτηση από dataset.}
Καμία μέθοδος δεν αναμένεται να κυριαρχεί σε όλα τα datasets. Η σχετική απόδοση εξαρτάται κυρίως από τη φύση της κατανομής δυσκολίας ερωτημάτων — μεταβλητή που δεν ελέγχεται από τον αλγόριθμο αλλά από τα χαρακτηριστικά του εκάστοτε dataset. Αυτή η υπόθεση έχει άμεσες πρακτικές συνέπειες: η επιλογή μεθόδου για παραγωγικό σύστημα πρέπει να γίνεται με βάση ανάλυση της κατανομής των πραγματικών ερωτημάτων.

    \item \textbf{\selectlanguage{english}GloVe-100\selectlanguage{greek}:} Πάνω από ένα εκατομμύριο διανύσματα λέξεων, 100 διαστάσεων. Το σύνολο αυτό παρουσιάζει ιδιαίτερη δυσκολία όχι λόγω μεγέθους, αλλά λόγω της σχετικά ομοιόμορφης κατανομής των διανυσμάτων στον χώρο: όταν δεν υπάρχει έντονη ομαδοποίηση (clustering), οι πλησιέστεροι γείτονες ενός ερωτήματος δεν απέχουν αισθητά από τους υπόλοιπους κόμβους, οπότε η greedy πλοήγηση έχει λιγότερη δομή να εκμεταλλευτεί και χρειάζεται μεγαλύτερο $ef_{\text{search}}$ για ικανοποιητικό \selectlanguage{english}recall\selectlanguage{greek}. Αξιολογείται με ευκλείδεια απόσταση στα αρχικά (μη κανονικοποιημένα) διανύσματα.

    \item \textbf{\selectlanguage{english}SIFT-1M\selectlanguage{greek}:} Ένα εκατομμύριο διανύσματα SIFT χαρακτηριστικών εικόνας, 128 διαστάσεων. Επιλέχθηκε για την αξιολόγηση κλιμακωσιμότητας, καθώς η κλίμακα αυτή αντιστοιχεί σε τυπικές διανυσματικές βάσεις παραγωγικής χρήσης.
\end{itemize}

\subsection{Μετρικές Αξιολόγησης}

Οι δύο κεντρικές μετρικές αξιολόγησης είναι το \selectlanguage{english}Recall@k\selectlanguage{greek} και το \selectlanguage{english}QPS\selectlanguage{greek}. Από κοινού αποτυπώνουν τον συμβιβασμό ακρίβειας--ταχύτητας: δεν υπάρχει μονοδιάστατη βελτιστότητα, και κάθε μέθοδος αξιολογείται στο πλαίσιο αυτής της αντιπαράθεσης.

\subsubsection{\selectlanguage{english}Recall@k}
Το \selectlanguage{english}Recall@k\selectlanguage{greek} μετράει τι ποσοστό από τους πραγματικούς $k$ κοντινότερους γείτονες επέστρεψε ο αλγόριθμος:
\[
\text{Recall@}k = \frac{1}{n_q} \sum_{i=1}^{n_q} \frac{|\hat{N}_k(q_i) \cap N_k(q_i)|}{k}
\]
όπου $N_k(q_i)$ είναι οι πραγματικοί $k$ γείτονες του ερωτήματος $q_i$ (υπολογισμένοι εξαντλητικά) και $\hat{N}_k(q_i)$ αυτοί που επέστρεψε ο αλγόριθμος. Τιμή 1.0 σημαίνει τέλεια ακρίβεια.

\subsubsection{\selectlanguage{english}QPS (Queries Per Second)}
Το \selectlanguage{english}QPS\selectlanguage{greek} εκφράζει πόσα ερωτήματα επεξεργάζεται ο αλγόριθμος ανά δευτερόλεπτο. Όλες οι μετρήσεις έγιναν σε \selectlanguage{english}single-threaded\selectlanguage{greek} λειτουργία, ώστε η σύγκριση να μην επηρεάζεται από διαφορές στην παραλληλοποίηση --- που θα έκαναν αναπόφευκτα τη σύγκριση άδικη.

\subsection{Πειραματική Διαδικασία}

\subsubsection{Περιβάλλον Εκτέλεσης}
Τα πειράματα εκτελέστηκαν σε μηχάνημα με \selectlanguage{english}Apple M1\selectlanguage{greek}, σε \selectlanguage{english}single-threaded\selectlanguage{greek} λειτουργία. Οι υλοποιήσεις \selectlanguage{english}Baseline-HNSW, DARTH, PiP\selectlanguage{greek} και \selectlanguage{english}Ada-ef\selectlanguage{greek} αναπτύχθηκαν από εμάς στο πλαίσιο της παρούσας εργασίας, βασισμένες στις αντίστοιχες δημοσιεύσεις, και είναι γραμμένες σε \selectlanguage{english}C++\selectlanguage{greek} με έκθεση στην \selectlanguage{english}Python\selectlanguage{greek} μέσω \selectlanguage{english}pybind11\selectlanguage{greek}. Για τη σύγκριση με βιομηχανική υλοποίηση, συμπεριλάβαμε τη \selectlanguage{english}faiss-hnsw\selectlanguage{greek} από την επίσημη βιβλιοθήκη \selectlanguage{english}FAISS\selectlanguage{greek} \cite{faiss2017}, με \selectlanguage{english}OpenMP\selectlanguage{greek} απενεργοποιημένο (\selectlanguage{english}omp\_set\_num\_threads(1)\selectlanguage{greek}), ώστε να μην έχει πλεονέκτημα παραλληλοποίησης έναντι των υπολοίπων μεθόδων.

\subsubsection{Πλέγμα Παραμέτρων}
Για κάθε μέθοδο εκτελέστηκε πλήρης σάρωση παραμέτρων:
\begin{itemize}
    \item $M \in \{8, 16\}$: μέγιστος αριθμός γειτόνων ανά κόμβο
    \item $ef_{\text{construction}} \in \{128, 200\}$: εύρος κατά την κατασκευή του \selectlanguage{english}index\selectlanguage{greek}
    \item $ef_{\text{search}} \in \{10, 20, 50, 100, 150, 200\}$: εύρος αναζήτησης (\selectlanguage{english}Baseline, DARTH, PiP, faiss-hnsw\selectlanguage{greek})
    \item $k \in \{10, 20\}$: αριθμός ζητούμενων γειτόνων
\end{itemize}

\subsubsection{Παράμετροι ανά Μέθοδο}
\begin{itemize}
    \item \textbf{\selectlanguage{english}DARTH\selectlanguage{greek}:} Στόχος ανάκλησης $R_t = 0.90$. Το μοντέλο πρόβλεψης (\selectlanguage{english}LightGBM\selectlanguage{greek}) εκπαιδεύτηκε ανά συνδυασμό $(dataset, M, ef_{\text{construction}}, k)$ με $n_{estimators} = 100$ χρησιμοποιώντας σύνολο εκπαίδευσης από 3.000--5.000 ερωτήματα.
    \item \textbf{\selectlanguage{english}PiP\selectlanguage{greek}:} Παράμετρος $\gamma = 95\%$ για \selectlanguage{english}SIFT Small, Fashion-MNIST\selectlanguage{greek} και $\gamma = 99\%$ για \selectlanguage{english}MS MARCO Text, GloVe-100, SIFT-1M\selectlanguage{greek}, $\delta = 20$.
    \item \textbf{\selectlanguage{english}Ada-ef\selectlanguage{greek}:} Στόχος $R_t = 0.90$, λειτουργεί με \selectlanguage{english}cosine similarity\selectlanguage{greek} μετά από \selectlanguage{english}L2\selectlanguage{greek} κανονικοποίηση των διανυσμάτων. Το σύνολο βαθμονόμησης (\selectlanguage{english}calibration\selectlanguage{greek}) χτίζεται για $ef \in \{20, 50, 75, 100, 150, 200, 300\}$.
\end{itemize}

\subsection{Ανάλυση ανά Dataset}

Τα συγκεντρωτικά αποτελέσματα κρύβουν ενδιαφέρουσες συμπεριφορές που αναδύονται μόνο όταν εξετάζουμε κάθε dataset ξεχωριστά.

\textbf{\selectlanguage{english}SIFT Small\selectlanguage{greek}.} Σε αυτό το μικρό σύνολο, όλες οι μέθοδοι επιτυγχάνουν \selectlanguage{english}Recall@10\selectlanguage{greek} $> 0.97$ για $ef_{\text{search}} \geq 20$. Οι διαφορές σε \selectlanguage{english}QPS\selectlanguage{greek} αντανακλούν κυρίως το \selectlanguage{english}overhead\selectlanguage{greek} κάθε μεθόδου παρά αλγοριθμική υπεροχή: το κόστος επίκλησης του \selectlanguage{english}LightGBM\selectlanguage{greek} στη \selectlanguage{english}DARTH\selectlanguage{greek} είναι αισθητό όταν η αναζήτηση η ίδια έχει μόλις λίγα βήματα. Η \selectlanguage{english}faiss-hnsw\selectlanguage{greek} επιτυγχάνει $\approx 2\times$ υψηλότερο \selectlanguage{english}QPS\selectlanguage{greek} έναντι της \selectlanguage{english}Baseline\selectlanguage{greek}, επιβεβαιώνοντας ότι οι βελτιστοποιήσεις χαμηλού επιπέδου έχουν μεγαλύτερο αντίκτυπο από την αλγοριθμική στρατηγική σε αυτή την κλίμακα.

\textbf{\selectlanguage{english}Fashion-MNIST\selectlanguage{greek}.} Η υψηλή διάσταση (784) δοκιμάζει το \selectlanguage{english}HNSW\selectlanguage{greek}: για $ef_{\text{search}} = 20$, το \selectlanguage{english}Recall@10\selectlanguage{greek} της \selectlanguage{english}Baseline\selectlanguage{greek} κυμαίνεται γύρω στο 0.91. Η \selectlanguage{english}DARTH\selectlanguage{greek} αναδεικνύεται εδώ: για $ef_{\text{search}} > 50$ κερδίζει βήματα τερματισμού χωρίς ουσιαστική πτώση ανάκλησης, διότι σημαντικό μέρος των ερωτημάτων συγκλίνει γρήγορα. Το μοντέλο εκμεταλλεύεται αξιόπιστα αυτή την ετερογένεια.

\textbf{\selectlanguage{english}MS MARCO Text\selectlanguage{greek}.} Τα \selectlanguage{english}text embeddings\selectlanguage{greek} εμφανίζουν ισχυρή αρκουδοειδή δομή (\selectlanguage{english}clustering\selectlanguage{greek}): ερωτήματα που ανήκουν σε θεματικές ομάδες τείνουν να έχουν παρόμοιους γείτονες. Αυτό ευνοεί τη \selectlanguage{english}DARTH\selectlanguage{greek} (σαφή σήματα σύγκλισης) αλλά εκθέτει περισσότερο την \selectlanguage{english}PiP\selectlanguage{greek}: σε \selectlanguage{english}clustered\selectlanguage{greek} δεδομένα, το \selectlanguage{english}top-$k$\selectlanguage{greek} σύνολο σταθεροποιείται πρόωρα εντός μιας θεματικής ομάδας, ενεργοποιώντας το κριτήριο κορεσμού πριν εξερευνηθούν γειτονικές ομάδες που περιέχουν τους πραγματικούς κοντινότερους γείτονες.

\textbf{\selectlanguage{english}GloVe-100\selectlanguage{greek}.} Αποτελεί το πιο απαιτητικό σύνολο για τις προσαρμοστικές μεθόδους. Η κατανομή των διανυσμάτων λέξεων είναι σχεδόν ομοιόμορφη σε μεγάλο τμήμα του λεξιλογίου, άρα δεν υπάρχει διαφοροποίηση δυσκολίας ερωτημάτων που να μπορούν να εκμεταλλευτούν οι προσαρμοστικές στρατηγικές. Το \selectlanguage{english}Recall@10\selectlanguage{greek} της \selectlanguage{english}Baseline\selectlanguage{greek} για $ef_{\text{search}}=100$ φτάνει μόλις το 0.739, ενδεικτικό της εγγενούς δυσκολίας του συνόλου. Η \selectlanguage{english}Ada-ef\selectlanguage{greek} επιτυγχάνει 0.871 με \selectlanguage{english}cosine similarity\selectlanguage{greek} --- βελτίωση που αντικατοπτρίζει κυρίως τη διαφορά στη μετρική, όχι αλγοριθμική υπεροχή.

\textbf{\selectlanguage{english}SIFT-1M\selectlanguage{greek}.} Σε κλίμακα εκατομμυρίου διανυσμάτων, το κόστος κατασκευής index αυξάνεται δραματικά και οι διαφορές σε \selectlanguage{english}QPS\selectlanguage{greek} αρχίζουν να αντικατοπτρίζουν αμιγώς αλγοριθμικές διαφορές. \subsection{Μελλοντική Εργασία}

Κάποια ερωτήματα που προέκυψαν κατά τη διεξαγωγή της εργασίας παραμένουν ανοιχτά και αξίζουν περαιτέρω διερεύνηση:

\begin{itemize}
    \item \textbf{Προσαρμοστικό \selectlanguage{english}DARTH\selectlanguage{greek} \selectlanguage{english}online\selectlanguage{greek}.} Το μοντέλο \selectlanguage{english}DARTH\selectlanguage{greek} εκπαιδεύτηκε \selectlanguage{english}offline\selectlanguage{greek} με σταθερό σύνολο χαρακτηριστικών ανά συνδυασμό $(dataset, M, ef_C, k)$. Εναλλακτικά, η αντικατάσταση του \selectlanguage{english}LightGBM\selectlanguage{greek} με ελαφρύτερο νευρωνικό δίκτυο (π.χ. ένα \selectlanguage{english}MLP\selectlanguage{greek} δύο στρωμάτων με \selectlanguage{english}ONNX\selectlanguage{greek} εξαγωγή) θα μπορούσε να μειώσει την επιβάρυνση επίκλησης σε χαμηλά $ef_{\text{search}}$.

    \item \textbf{Δυναμικές βάσεις δεδομένων.} Όλες οι μέθοδοι αξιολογήθηκαν σε στατικά datasets. Σε παραγωγικές βάσεις δεδομένων διανυσμάτων, εισαγωγές και διαγραφές συμβαίνουν συνεχώς. Πώς επηρεάζεται η ποιότητα του γράφου --- και κατ' επέκταση η απόδοση των προσαρμοστικών στρατηγικών --- όταν μέρος των κόμβων αντικαθίσταται, παραμένει ανοιχτό. Ο \selectlanguage{english}HNSW\selectlanguage{greek} υποστηρίζει δυναμικές εισαγωγές, αλλά δεν υποστηρίζει αποδοτικές διαγραφές χωρίς επαναδόμηση.

    \item \textbf{Κλιμάκωση σε δισεκατομμύρια διανύσματα.} Η αξιολόγησή μας έφτασε μέχρι $10^6$ διανύσματα (\selectlanguage{english}SIFT-1M\selectlanguage{greek}), όπου η \selectlanguage{english}DARTH\selectlanguage{greek} κλιμακώθηκε ικανοποιητικά. Ωστόσο, τα παραγωγικά συστήματα μεγάλης κλίμακας (π.χ. \selectlanguage{english}Pinecone, Weaviate, Milvus\selectlanguage{greek}) διαχειρίζονται δισεκατομμύρια διανύσματα. Σε αυτή την κλίμακα, το κόστος κατασκευής index και εκπαίδευσης predictor αναδεικνύεται ως κρίσιμο εμπόδιο: για $N = 10^9$, ακόμα και η απλή εισαγωγή σε \selectlanguage{english}HNSW\selectlanguage{greek} απαιτεί εβδομάδες σε κανονικό hardware.

    \item \textbf{Ανάλυση κορεσμού \selectlanguage{english}PiP\selectlanguage{greek}.} Ο κορεσμός ανάκλησης που παρατηρήσαμε στην \selectlanguage{english}PiP\selectlanguage{greek} --- \selectlanguage{english}Recall@10\selectlanguage{greek} $= 0.944$ για $ef_{\text{search}} \geq 100$ στο \selectlanguage{english}SIFT-1M\selectlanguage{greek} ενώ η \selectlanguage{english}Baseline\selectlanguage{greek} επιτυγχάνει 0.985 --- παραμένει αναλυτικά ανεξήγητος. Πιθανή αιτία είναι ότι σε ορισμένα ερωτήματα το \selectlanguage{english}top-$k$\selectlanguage{greek} σταθεροποιείται πρόωρα σε τοπικό ελάχιστο του γράφου, ενεργοποιώντας τον μηχανισμό \selectlanguage{english}patience\selectlanguage{greek} πριν εξερευνηθούν «γέφυρες» σύνδεσης διαφορετικών τμημάτων. Μια αναλυτική χαρτογράφηση των ερωτημάτων που αποτυγχάνουν --- διάκριση μεταξύ πρόωρου τερματισμού και ελλιπούς εξερεύνησης --- θα μπορούσε να προσδιορίσει αν το πρόβλημα εντοπίζεται στο κριτήριο $\phi_{h,l} \geq \gamma$ ή στην τιμή του $\Delta$.

    \item \textbf{Συνδυασμός στρατηγικών.} Οι τρεις μέθοδοι εξετάστηκαν ανεξάρτητα. Ένας φυσικός επόμενος βηματισμός είναι η διερεύνηση συνδυασμών: για παράδειγμα, \selectlanguage{english}DARTH\selectlanguage{greek} για πρόωρο τερματισμό συνδυαζόμενη με \selectlanguage{english}SIMD\selectlanguage{greek} βελτιστοποιήσεις τύπου \selectlanguage{english}FAISS\selectlanguage{greek} για ταχύτερο υπολογισμό απόστασης. Τέτοιοι συνδυασμοί θα μπορούσαν να αξιοποιήσουν τα πλεονεκτήματα και των δύο κατευθύνσεων.
\end{itemize}

// Returns top-ef_search candidates from level 0 starting at entry point ep
std::priority_queue<std::pair<float,int>> HNSW::searchLayer0(
    const float* query, int ep, int ef) {

    std::priority_queue<std::pair<float,int>> topSet;   // max-heap (results)
    std::priority_queue<std::pair<float,int>,
        std::vector<std::pair<float,int>>,
        std::greater<>> candidates;                     // min-heap (to expand)

    float d0 = distance(query, getVector(ep));
    dist_counter_++;                                    // atomic instrumentation
    topSet.push({d0, ep});
    candidates.push({d0, ep});
    visited_.mark(ep);

    while (!candidates.empty()) {
        auto [dc, c] = candidates.top(); candidates.pop();
        // Termination: best candidate is worse than worst result
        if (dc > topSet.top().first) break;

        for (int nb : neighbors_[0][c]) {
            if (visited_.check(nb)) continue;
            visited_.mark(nb);
            float dn = distance(query, getVector(nb));
            dist_counter_++;
            if (dn < topSet.top().first || (int)topSet.size() < ef) {
                topSet.push({dn, nb});
                candidates.push({dn, nb});
                if ((int)topSet.size() > ef) topSet.pop();
            }
        }
    }
    return topSet;
}

// Feature extraction at checkpoint t (called every delta steps)
std::array<float, 5> DARTH::extractFeatures(
    const TopSet& W, const TopSet& W_prev, int step) {

    float d_min_cur  = W.worstDist();      // k-th distance in top set
    float d_min_prev = W_prev.worstDist();
    float d1         = W.bestDist();       // closest distance seen so far

    return {
        d_min_prev - d_min_cur,            // delta_d_min: improvement
        W.stabilityVsSnapshot(W_prev),     // s_topk: fraction unchanged
        (float)(step - last_improvement_), // delta_t_last
        (float)W.size() / ef_search_,      // r_W: fill ratio
        (d_min_cur > 0) ? d_min_cur / d1 : 1.0f  // d_k / d_1 ratio
    };
}

bool DARTH::shouldTerminate(const std::array<float,5>& feats) {

// Inside search_layer_pip() at level 0
std::unordered_set<int> prev_topk;
int saturation_counter = 0;

while (!C.empty()) {
    // ... standard HNSW expansion step ...

    // PiP saturation check: phi_h,l(q) = 100 * |prev_topk ∩ curr_topk| / k
    std::vector<int> curr = heap_topk_ids(Kheap, k);
    if ((int)curr.size() == k) {
        std::unordered_set<int> curr_set(curr.begin(), curr.end());
        if (!prev_topk.empty()) {
            int overlap = 0;
            for (int x : curr_set)
                if (prev_topk.count(x)) ++overlap;
            double phi = 100.0 * overlap / k;
            if (phi >= pip_gamma_) ++saturation_counter;
            else saturation_counter = 0;
            if (saturation_counter >= pip_delta_) break; // patience reached
        }
        prev_topk = std::move(curr_set);
    }
}

// At query time: greedy walk on upper layers gives similarity score s_hat
float AdaEf::estimateSimilarity(const float* query, int ep) {
    float sim_sum = 0.0f;
    int   count   = 0;
    int   cur     = ep;
    for (int l = top_layer_; l >= 1; --l) {
        bool improved = true;
        while (improved) {
            improved = false;
            for (int nb : neighbors_[l][cur]) {
                float s = cosineSimilarity(query, getVector(nb));
                sim_sum += s; count++;
                if (s > cosineSimilarity(query, getVector(cur)))
                    { cur = nb; improved = true; }
            }
        }
    }
    return (count > 0) ? sim_sum / count : 0.0f;
}

// Select minimum ef that achieves recall_target for this similarity bucket
int AdaEf::selectEf(float s_hat) {
    int s_bucket = static_cast<int>(s_hat * lut_bins_);
    for (int ef : ef_candidates_) {              // sorted ascending
        if (lut_[ef][s_bucket] >= recall_target_) return ef;
    }
    return ef_candidates_.back();                // fallback: maximum ef
}
